import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans
import warnings
warnings.filterwarnings('ignore')

print("\n" + "="*70)
print("⚙️ ACTIVIDAD 6: PREDICTIVE MAINTENANCE - EXTRACCIÓN DE VALOR VDRER")
print("="*70)

# ===== FASE 1: VALIDAR =====
df = pd.read_csv('predictive_maintenance.csv')
print(f"✅ Dataset cargado: {df.shape[0]:,} filas × {df.shape[1]} columnas")

# Overview del sistema industrial
print(f"\n🏭 OVERVIEW INDUSTRIAL SYSTEM:")
print(f"🔧 Equipment Units: {df['UDI'].nunique():,}")
print(f"📦 Product Types: {df['Product ID'].nunique():,}")
print(f"⚡ Equipment Classes: {df['Type'].value_counts().to_dict()}")
print(f"💥 Overall Failure Rate: {df['Target'].mean():.3f} ({df['Target'].mean()*100:.1f}%)")

# Sensor data summary
print(f"\n📊 SENSOR DATA SUMMARY:")
sensor_cols = ['Air temperature [K]', 'Process temperature [K]', 
               'Rotational speed [rpm]', 'Torque [Nm]', 'Tool wear [min]']
for col in sensor_cols:
    print(f"   {col}: {df[col].min():.1f} - {df[col].max():.1f} (mean: {df[col].mean():.1f})")

# ===== FASE 2: DESCUBRIR =====
print(f"\n🔍 FASE 2: DESCUBRIR - Variables 'inútiles' industriales")

insights_found = []

# UDI patterns analysis
print(f"\n📊 ANÁLISIS UDI (Unique Device Identifier):")
print(f"Rango: {df['UDI'].min()} - {df['UDI'].max()}")
udi_gaps = df['UDI'].diff().describe()
print(f"Gaps en secuencia UDI: mean={udi_gaps['mean']:.1f}, std={udi_gaps['std']:.1f}")

# Product ID structure analysis
print(f"\n📊 ANÁLISIS PRODUCT ID:")
print(f"Formato típico: {df['Product ID'].iloc[0]}")
product_prefixes = df['Product ID'].str[0].value_counts()
print(f"Prefijos encontrados: {product_prefixes.to_dict()}")

# ===== FASE 3: REUTILIZAR =====
print(f"\n♻️ FASE 3: REUTILIZAR - Extrayendo valor de códigos industriales")

# Equipment installation cohorts (basado en UDI)
df['installation_cohort'] = pd.qcut(df['UDI'], q=5, labels=['Gen1', 'Gen2', 'Gen3', 'Gen4', 'Gen5'])

# Product family extraction (basado en Product ID)
df['product_family'] = df['Product ID'].str[0]

# Equipment age proxy (basado en UDI como proxy de installation order)
df['equipment_age_proxy'] = (df['UDI'] - df['UDI'].min()) / (df['UDI'].max() - df['UDI'].min())

# Failure risk by type
failure_by_type = df.groupby('Type')['Target'].agg(['mean', 'count'])
print(f"\n⚠️ FAILURE RATE BY EQUIPMENT TYPE:")
for eq_type in failure_by_type.index:
    failure_rate = failure_by_type.loc[eq_type, 'mean']
    count = failure_by_type.loc[eq_type, 'count']
    print(f"   Type {eq_type}: {failure_rate:.3f} failure rate ({count:,} units)")

insights_found.extend([
    "UDI revela equipment installation cohorts con diferentes reliability",
    "Product ID estructura indica product families con failure patterns",
    "Type classification correlaciona con failure rates específicos"
])

# ===== FASE 4: ENGINEER =====
print(f"\n🔧 FASE 4: ENGINEER - Creando features de reliability")

new_features_created = []

# Equipment Reliability Index
# Combinar múltiples factores de reliability
df['reliability_index'] = (
    (1 - df['Target']) * 0.4 +  # No failure = high reliability
    (1 - df['equipment_age_proxy']) * 0.2 +  # Newer = more reliable
    (df['Tool wear [min]'] / df['Tool wear [min]'].max()) * -0.2 +  # Less wear = more reliable
    (df['Type'].map({'L': 0.6, 'M': 0.8, 'H': 1.0})) * 0.2  # Higher quality = more reliable
)

# Maintenance Efficiency Score
# Basado en operating conditions vs failure
df['operating_stress'] = (
    (df['Air temperature [K]'] - df['Air temperature [K]'].mean()).abs() / df['Air temperature [K]'].std() +
    (df['Process temperature [K]'] - df['Process temperature [K]'].mean()).abs() / df['Process temperature [K]'].std() +
    (df['Torque [Nm]'] - df['Torque [Nm]'].mean()).abs() / df['Torque [Nm]'].std()
) / 3

df['maintenance_efficiency'] = np.where(
    df['Target'] == 0,  # No failure
    1 / (df['operating_stress'] + 0.1),  # Higher efficiency under stress
    0.1  # Low efficiency if failed
)

# Failure Risk Assessment
# Predictive score basado en múltiples variables
scaler = StandardScaler()
risk_features = ['Air temperature [K]', 'Process temperature [K]', 
                'Rotational speed [rpm]', 'Torque [Nm]', 'Tool wear [min]']
risk_scaled = scaler.fit_transform(df[risk_features])

# Crear score de riesgo basado en distancia de condiciones normales
df['failure_risk_score'] = np.sqrt(np.sum(risk_scaled**2, axis=1))

# Equipment Lifecycle Stage
df['lifecycle_stage'] = pd.cut(df['equipment_age_proxy'], 
                              bins=[0, 0.2, 0.5, 0.8, 1.0],
                              labels=['NewInstall', 'Operating', 'Mature', 'EndOfLife'])

# Equipment Criticality Assessment
df['criticality_score'] = (
    df['failure_risk_score'] * 0.4 +
    (df['Type'].map({'L': 1, 'M': 2, 'H': 3})) * 0.3 +
    df['operating_stress'] * 0.3
)

df['criticality_level'] = pd.cut(df['criticality_score'],
                                bins=[0, 1, 2, 3, float('inf')],
                                labels=['Low', 'Medium', 'High', 'Critical'])

new_features_created.extend([
    'reliability_index', 'maintenance_efficiency', 'failure_risk_score',
    'lifecycle_stage', 'criticality_level', 'operating_stress'
])

print(f"✅ Features industriales creadas: {len(new_features_created)}")

# Performance analysis por cohort
cohort_analysis = df.groupby('installation_cohort').agg({
    'Target': 'mean',
    'reliability_index': 'mean',
    'maintenance_efficiency': 'mean'
}).round(3)

print(f"\n📈 PERFORMANCE BY INSTALLATION COHORT:")
for cohort in cohort_analysis.index:
    failure_rate = cohort_analysis.loc[cohort, 'Target']
    reliability = cohort_analysis.loc[cohort, 'reliability_index']
    efficiency = cohort_analysis.loc[cohort, 'maintenance_efficiency']
    print(f"   {cohort}: Failure={failure_rate:.3f}, Reliability={reliability:.3f}, Efficiency={efficiency:.3f}")

# ===== FASE 5: REPORTAR =====
print(f"\n📋 FASE 5: REPORTAR - Exportando dataset industrial enriquecido")

print(f"\n📊 REPORTE FINAL INDUSTRIAL:")
print(f"   💎 Insights de reliability extraídos: {len(insights_found)}")
print(f"   ⚙️ Features de maintenance creadas: {len(new_features_created)}")
print(f"   📈 Dimensiones finales: {df.shape}")

# Equipment criticality distribution
criticality_dist = df['criticality_level'].value_counts()
print(f"\n🚨 EQUIPMENT CRITICALITY DISTRIBUTION:")
for level in ['Low', 'Medium', 'High', 'Critical']:
    if level in criticality_dist.index:
        count = criticality_dist[level]
        pct = count / len(df) * 100
        print(f"   {level}: {count:,} equipos ({pct:.1f}%)")

# Lifecycle stage analysis
lifecycle_dist = df['lifecycle_stage'].value_counts()
print(f"\n📅 EQUIPMENT LIFECYCLE DISTRIBUTION:")
for stage in lifecycle_dist.index:
    count = lifecycle_dist[stage]

    print(f"   {stage}: {count:,} equipos ({pct:.1f}%)")

# Export enhanced dataset
try:

except Exception as e:

library(tidyverse)
library(corrplot)

# Cargar dataset enriquecido
predictive_data <- read_csv("predictive_maintenance_value_extracted_python.csv")

cat("⚙️ PREDICTIVE MAINTENANCE STATISTICAL VALIDATION\n")
cat("================================================\n")

# Verificar features VDRER
vdrer_features <- c("reliability_index", "maintenance_efficiency", 
                    "failure_risk_score", "criticality_level", "lifecycle_stage")

for(feature in vdrer_features) {
  if(feature %in% names(predictive_data)) {
    cat("✅", feature, "- Type:", class(predictive_data[[feature]])[1], "\n")
  } else {
    cat("❌", feature, "- NOT FOUND\n")
  }
}

# TEST 1: ANOVA - Equipment Type vs Reliability
if("Type" %in% names(predictive_data) && "reliability_index" %in% names(predictive_data)) {
  
  anova_data <- predictive_data %>%
    filter(!is.na(Type), !is.na(reliability_index), is.finite(reliability_index))
  
  if(nrow(anova_data) > 0) {
    anova_result <- aov(reliability_index ~ Type, data = anova_data)
    anova_summary <- summary(anova_result)
    
    cat("\n📊 ANOVA - Reliability Index por Equipment Type:\n")
    cat("F-statistic:", round(anova_summary[[1]][1,4], 3), "\n")
    cat("p-value:", format(anova_summary[[1]][1,5], scientific = TRUE), "\n")
    
    if(anova_summary[[1]][1,5] < 0.05) {
      cat("✅ Diferencias significativas entre equipment types\n")
      
      # Tukey HSD para comparaciones múltiples
      tukey_result <- TukeyHSD(anova_result)
      cat("📈 Comparaciones Type por reliability:\n")
      print(tukey_result)
    }
  }
}

# TEST 2: Installation Cohort Performance
if("installation_cohort" %in% names(predictive_data)) {
  
  cohort_analysis <- predictive_data %>%
    group_by(installation_cohort) %>%
    summarise(
      avg_reliability = mean(reliability_index, na.rm = TRUE),
      failure_rate = mean(Target, na.rm = TRUE),
      equipment_count = n()
    ) %>%
    arrange(desc(avg_reliability))
  
  cat("\n📊 INSTALLATION COHORT ANALYSIS:\n")
  print(cohort_analysis)
  
  # Test trend en cohorts (correlación ordinal)
  cohort_numeric <- as.numeric(factor(predictive_data$installation_cohort, 
                                     levels = c("Gen1", "Gen2", "Gen3", "Gen4", "Gen5")))
  cor_test <- cor.test(cohort_numeric, predictive_data$reliability_index, method = "spearman")
  
  cat("\nSpearman correlation (Cohort vs Reliability):", round(cor_test$estimate, 3), "\n")
  cat("p-value:", format(cor_test$p.value, scientific = TRUE), "\n")
  
  if(cor_test$p.value < 0.05) {
    cat("✅ Trend significativo en reliability por installation cohort\n")
  }
}

# TEST 3: Survival Analysis - Equipment Lifecycle
if("lifecycle_stage" %in% names(predictive_data)) {
  
  # Crear survival object usando equipment age proxy y failure
  survival_data <- predictive_data %>%
    mutate(
      time_to_event = equipment_age_proxy * 1000,  # Scale for analysis
      event = Target  # Failure event
    ) %>%
    filter(!is.na(lifecycle_stage), is.finite(time_to_event))
  
  if(nrow(survival_data) > 100) {
    # Kaplan-Meier por lifecycle stage
    surv_object <- Surv(survival_data$time_to_event, survival_data$event)
    km_fit <- survfit(surv_object ~ lifecycle_stage, data = survival_data)
    
    cat("\n⏱️ SURVIVAL ANALYSIS por Lifecycle Stage:\n")
    print(km_fit)
    
    # Log-rank test
    if(length(unique(survival_data$lifecycle_stage)) > 1) {
      logrank_test <- survdiff(surv_object ~ lifecycle_stage, data = survival_data)
      p_value <- 1 - pchisq(logrank_test$chisq, df = length(logrank_test$n) - 1)
      cat("Log-rank test p-value:", format(p_value, scientific = TRUE), "\n")
      
      if(p_value < 0.05) {
        cat("✅ Diferencias significativas en survival por lifecycle stage\n")
      }
    }
  }
}

# TEST 4: Equipment Clustering Validation
if("reliability_index" %in% names(predictive_data) && "maintenance_efficiency" %in% names(predictive_data)) {
  
  # Preparar datos para clustering
  cluster_features <- predictive_data %>%
    select(reliability_index, maintenance_efficiency, failure_risk_score) %>%
    na.omit() %>%
    filter(across(everything(), is.finite))
  
  if(nrow(cluster_features) > 100) {
    # K-means clustering
    cluster_scaled <- scale(cluster_features)
    
    set.seed(42)
    kmeans_result <- kmeans(cluster_scaled, centers = 4, nstart = 25)
    
    cat("\n🎯 EQUIPMENT CLUSTERING VALIDATION:\n")
    cat("Clusters identificados:", max(kmeans_result$cluster), "\n")
    
    # Silhouette analysis
    if(require(cluster, quietly = TRUE)) {
      sil_analysis <- silhouette(kmeans_result$cluster, dist(cluster_scaled))
      avg_sil_width <- mean(sil_analysis[, 3])
      cat("Silhouette Width:", round(avg_sil_width, 3), "\n")
      
      if(avg_sil_width > 0.5) {
        cat("✅ Clustering robusto - Equipment groups bien definidos\n")
