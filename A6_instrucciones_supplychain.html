print("\n🔧 FASE 4: ENGINEER - Creando features de supply chain con valor operacional")

new_supply_features = []

# Feature 1: Supply Chain Efficiency Index
try:
    # Combinar métricas de tiempo y costo para crear índice de eficiencia
    # Normalizar cada métrica (0-1) donde más alto = mejor
    
    # Normalizar lead times (invertir: menos tiempo = mejor)
    df['lead_time_norm'] = 1 - ((df['Lead times'] - df['Lead times'].min()) / 
                                (df['Lead times'].max() - df['Lead times'].min() + 0.001))
    
    # Normalizar shipping times (invertir: menos tiempo = mejor)
    df['shipping_time_norm'] = 1 - ((df['Shipping times'] - df['Shipping times'].min()) / 
                                    (df['Shipping times'].max() - df['Shipping times'].min() + 0.001))
    
    # Normalizar manufacturing lead time (invertir: menos tiempo = mejor)
    df['mfg_lead_time_norm'] = 1 - ((df['Manufacturing lead time'] - df['Manufacturing lead time'].min()) / 
                                   (df['Manufacturing lead time'].max() - df['Manufacturing lead time'].min() + 0.001))
    
    # Crear índice compuesto de eficiencia
    df['supply_chain_efficiency_index'] = (
        df['lead_time_norm'] * 0.4 +
        df['shipping_time_norm'] * 0.3 +
        df['mfg_lead_time_norm'] * 0.3
    )
    
    print("✅ supply_chain_efficiency_index: Índice compuesto de eficiencia temporal")
    print(f"   Media: {df['supply_chain_efficiency_index'].mean():.3f}")
    print(f"   Interpretación: Más alto = supply chain más eficiente en tiempos")
    new_supply_features.append('supply_chain_efficiency_index')
except Exception as e:
    print(f"⚠️ Error creando supply_chain_efficiency_index: {e}")

# Feature 2: Cost-to-Revenue Ratio
try:
    # Calcular ratio de costos totales vs revenue generado
    df['total_costs'] = (df['Shipping costs'] + df['Manufacturing costs'] + df['Costs'])
    df['cost_to_revenue_ratio'] = df['total_costs'] / (df['Revenue generated'] + 0.001)  # Evitar división por 0
    
    print("\n✅ cost_to_revenue_ratio: Ratio de eficiencia de costos")
    print(f"   Media: {df['cost_to_revenue_ratio'].mean():.3f}")
    print(f"   Interpretación: Más bajo = mejor eficiencia de costos")
    new_supply_features.append('cost_to_revenue_ratio')
except Exception as e:
    print(f"⚠️ Error creando cost_to_revenue_ratio: {e}")

# Feature 3: Quality Performance Score
try:
    # Crear score de calidad basado en defect rates e inspection results
    
    # Normalizar defect rates (invertir: menos defectos = mejor)
    df['defect_rate_norm'] = 1 - ((df['Defect rates'] - df['Defect rates'].min()) / 
                                  (df['Defect rates'].max() - df['Defect rates'].min() + 0.001))
    
    # Score por inspection results
    inspection_scores = {'Pass': 1.0, 'Pending': 0.5, 'Fail': 0.0}
    df['inspection_score'] = df['Inspection results'].map(inspection_scores)
    
    # Score compuesto de calidad
    df['quality_performance_score'] = (
        df['defect_rate_norm'] * 0.6 +
        df['inspection_score'] * 0.4
    )
    
    print("\n✅ quality_performance_score: Score de performance de calidad")
    print(f"   Media: {df['quality_performance_score'].mean():.3f}")
    print(f"   Interpretación: Más alto = mejor calidad del producto")

    transport_efficiency = {'Air': 0.9, 'Road': 0.7, 'Rail': 0.6, 'Sea': 0.4}
    df['transport_efficiency_score'] = df['Transportation modes'].map(transport_efficiency)
    
    # Normalizar shipping costs (invertir: menos costo = mejor)
    df['shipping_cost_norm'] = 1 - ((df['Shipping costs'] - df['Shipping costs'].min()) / 
                                    (df['Shipping costs'].max() - df['Shipping costs'].min() + 0.001))
    
    # Índice de optimización logística

# ⚙️ FASE 4: ENGINEER - Crear Features Derivadas de Supply Chain
# ===================================================================

print("\n🔧 FASE 4: ENGINEER - Creando features de supply chain con valor operacional")

new_supply_features = []

# Feature 1: Supply Chain Efficiency Index
try:
    # Combinar métricas de tiempo y costo para crear índice de eficiencia
    # Normalizar cada métrica (0-1) donde más alto = mejor
    
    # Normalizar lead times (invertir: menos tiempo = mejor)
    df['lead_time_norm'] = 1 - ((df['Lead times'] - df['Lead times'].min()) / 
                                (df['Lead times'].max() - df['Lead times'].min() + 0.001))
    
    # Normalizar shipping times (invertir: menos tiempo = mejor)
    df['shipping_time_norm'] = 1 - ((df['Shipping times'] - df['Shipping times'].min()) / 
                                    (df['Shipping times'].max() - df['Shipping times'].min() + 0.001))
    
    # Normalizar manufacturing lead time (invertir: menos tiempo = mejor)
    df['mfg_lead_time_norm'] = 1 - ((df['Manufacturing lead time'] - df['Manufacturing lead time'].min()) / 
                                   (df['Manufacturing lead time'].max() - df['Manufacturing lead time'].min() + 0.001))
    
    # Crear índice compuesto de eficiencia
    df['supply_chain_efficiency_index'] = (
        df['lead_time_norm'] * 0.4 +
        df['shipping_time_norm'] * 0.3 +
        df['mfg_lead_time_norm'] * 0.3
    )
    
    print("✅ supply_chain_efficiency_index: Índice compuesto de eficiencia temporal")
    print(f"   Media: {df['supply_chain_efficiency_index'].mean():.3f}")
    print(f"   Interpretación: Más alto = supply chain más eficiente en tiempos")
    new_supply_features.append('supply_chain_efficiency_index')
except Exception as e:
    print(f"⚠️ Error creando supply_chain_efficiency_index: {e}")

# Feature 2: Cost-to-Revenue Ratio
try:
    # Calcular ratio de costos totales vs revenue generado
    df['total_costs'] = (df['Shipping costs'] + df['Manufacturing costs'] + df['Costs'])
    df['cost_to_revenue_ratio'] = df['total_costs'] / (df['Revenue generated'] + 0.001)  # Evitar división por 0
    
    print("\n✅ cost_to_revenue_ratio: Ratio de eficiencia de costos")
    print(f"   Media: {df['cost_to_revenue_ratio'].mean():.3f}")
    print(f"   Interpretación: Más bajo = mejor eficiencia de costos")
    new_supply_features.append('cost_to_revenue_ratio')
except Exception as e:
    print(f"⚠️ Error creando cost_to_revenue_ratio: {e}")

# Feature 3: Quality Performance Score
try:
    # Crear score de calidad basado en defect rates e inspection results
    
    # Normalizar defect rates (invertir: menos defectos = mejor)
    df['defect_rate_norm'] = 1 - ((df['Defect rates'] - df['Defect rates'].min()) / 
                                  (df['Defect rates'].max() - df['Defect rates'].min() + 0.001))
    
    # Score por inspection results
    inspection_scores = {'Pass': 1.0, 'Pending': 0.5, 'Fail': 0.0}
    df['inspection_score'] = df['Inspection results'].map(inspection_scores)
    
    # Score compuesto de calidad
    df['quality_performance_score'] = (
        df['defect_rate_norm'] * 0.6 +
        df['inspection_score'] * 0.4
    )
    
    print("\n✅ quality_performance_score: Score de performance de calidad")
    print(f"   Media: {df['quality_performance_score'].mean():.3f}")
    print(f"   Interpretación: Más alto = mejor calidad del producto")
    new_supply_features.append('quality_performance_score')
except Exception as e:
    print(f"⚠️ Error creando quality_performance_score: {e}")

# Feature 4: Logistics Optimization Index
try:
    # Combinar métricas de shipping y transportation para optimización logística
    
    # Crear score por transportation mode (asumiendo Air=más rápido pero caro, Sea=económico pero lento)
    transport_efficiency = {'Air': 0.9, 'Road': 0.7, 'Rail': 0.6, 'Sea': 0.4}
    df['transport_efficiency_score'] = df['Transportation modes'].map(transport_efficiency)
    
    # Normalizar shipping costs (invertir: menos costo = mejor)
    df['shipping_cost_norm'] = 1 - ((df['Shipping costs'] - df['Shipping costs'].min()) / 
                                    (df['Shipping costs'].max() - df['Shipping costs'].min() + 0.001))
    
    # Índice de optimización logística
    df['logistics_optimization_index'] = (
        df['transport_efficiency_score'] * 0.4 +
        df['shipping_cost_norm'] * 0.3 +
        df['shipping_time_norm'] * 0.3
    )
    
    print("\n✅ logistics_optimization_index: Índice de optimización logística")
    print(f"   Media: {df['logistics_optimization_index'].mean():.3f}")
    print(f"   Interpretación: Más alto = logística más optimizada")
    new_supply_features.append('logistics_optimization_index')
except Exception as e:
    print(f"⚠️ Error creando logistics_optimization_index: {e}")

# Feature 5: Supplier Risk Index
try:
    # Crear índice de riesgo basado en concentración de supplier
    supplier_counts = df['Supplier name'].value_counts()
    df['supplier_dependency'] = df['Supplier name'].map(supplier_counts)
    
    # Normalizar (más productos de un supplier = mayor riesgo de concentración)
    max_dependency = df['supplier_dependency'].max()
    df['supplier_risk_index'] = df['supplier_dependency'] / max_dependency
    
    print("\n✅ supplier_risk_index: Índice de riesgo de concentración de supplier")
    print(f"   Media: {df['supplier_risk_index'].mean():.3f}")
    print(f"   Interpretación: Más alto = mayor dependencia/riesgo de concentración")
    new_supply_features.append('supplier_risk_index')
except Exception as e:
    print(f"⚠️ Error creando supplier_risk_index: {e}")

try:
    # Relacionar production volumes con manufacturing lead time y costs
    df['production_efficiency_ratio'] = (
        df['Production volumes'] / 
        (df['Manufacturing lead time'] * df['Manufacturing costs'] + 0.001)

    print("\n✅ production_efficiency_ratio: Ratio de eficiencia de producción")
    print(f"   Media: {df['production_efficiency_ratio'].mean():.3f}")
    print(f"   Interpretación: Más alto = mayor eficiencia de producción")
    new_supply_features.append('production_efficiency_ratio')
except Exception as e:

# Feature 1: Supply Chain Efficiency Index
try:
    # Combinar métricas de tiempo y costo para crear índice de eficiencia
    # Normalizar cada métrica (0-1) donde más alto = mejor
    
    # Normalizar lead times (invertir: menos tiempo = mejor)
    df['lead_time_norm'] = 1 - ((df['Lead times'] - df['Lead times'].min()) / 
                                (df['Lead times'].max() - df['Lead times'].min() + 0.001))
    
    # Normalizar shipping times (invertir: menos tiempo = mejor)
    df['shipping_time_norm'] = 1 - ((df['Shipping times'] - df['Shipping times'].min()) / 
                                    (df['Shipping times'].max() - df['Shipping times'].min() + 0.001))
    
    # Normalizar manufacturing lead time (invertir: menos tiempo = mejor)
    df['mfg_lead_time_norm'] = 1 - ((df['Manufacturing lead time'] - df['Manufacturing lead time'].min()) / 
                                   (df['Manufacturing lead time'].max() - df['Manufacturing lead time'].min() + 0.001))
    
    # Crear índice compuesto de eficiencia
    df['supply_chain_efficiency_index'] = (
        df['lead_time_norm'] * 0.4 +
        df['shipping_time_norm'] * 0.3 +
        df['mfg_lead_time_norm'] * 0.3
    )
    
    print("✅ supply_chain_efficiency_index: Índice compuesto de eficiencia temporal")
    print(f"   Media: {df['supply_chain_efficiency_index'].mean():.3f}")
    print(f"   Interpretación: Más alto = supply chain más eficiente en tiempos")
    new_supply_features.append('supply_chain_efficiency_index')
except Exception as e:

try:

    df['total_costs'] = (df['Shipping costs'] + df['Manufacturing costs'] + df['Costs'])
    df['cost_to_revenue_ratio'] = df['total_costs'] / (df['Revenue generated'] + 0.001)  # Evitar división por 0

try:
    # Crear score de calidad basado en defect rates e inspection results
    
    # Normalizar defect rates (invertir: menos defectos = mejor)
    df['defect_rate_norm'] = 1 - ((df['Defect rates'] - df['Defect rates'].min()) / 
                                  (df['Defect rates'].max() - df['Defect rates'].min() + 0.001))
    
    # Score por inspection results
    inspection_scores = {'Pass': 1.0, 'Pending': 0.5, 'Fail': 0.0}
    df['inspection_score'] = df['Inspection results'].map(inspection_scores)
    
    # Score compuesto de calidad
    df['quality_performance_score'] = (
        df['defect_rate_norm'] * 0.6 +
        df['inspection_score'] * 0.4
    )
    
    print("\n✅ quality_performance_score: Score de performance de calidad")
    print(f"   Media: {df['quality_performance_score'].mean():.3f}")
    print(f"   Interpretación: Más alto = mejor calidad del producto")
    new_supply_features.append('quality_performance_score')
except Exception as e:

try:

    transport_efficiency = {'Air': 0.9, 'Road': 0.7, 'Rail': 0.6, 'Sea': 0.4}
    df['transport_efficiency_score'] = df['Transportation modes'].map(transport_efficiency)
    
    # Normalizar shipping costs (invertir: menos costo = mejor)
    df['shipping_cost_norm'] = 1 - ((df['Shipping costs'] - df['Shipping costs'].min()) / 
                                    (df['Shipping costs'].max() - df['Shipping costs'].min() + 0.001))
    
    # Índice de optimización logística

        df['shipping_cost_norm'] * 0.3 +

    print("\n✅ logistics_optimization_index: Índice de optimización logística")

print("\n🔧 FASE 4: ENGINEER - Creando features de supply chain con valor operacional")

new_supply_features = []

# Feature 1: Supply Chain Efficiency Index
try:
    # Combinar métricas de tiempo y costo para crear índice de eficiencia
    # Normalizar cada métrica (0-1) donde más alto = mejor
    
    # Normalizar lead times (invertir: menos tiempo = mejor)
    df['lead_time_norm'] = 1 - ((df['Lead times'] - df['Lead times'].min()) / 
                                (df['Lead times'].max() - df['Lead times'].min() + 0.001))
    
    # Normalizar shipping times (invertir: menos tiempo = mejor)
    df['shipping_time_norm'] = 1 - ((df['Shipping times'] - df['Shipping times'].min()) / 
                                    (df['Shipping times'].max() - df['Shipping times'].min() + 0.001))
    
    # Normalizar manufacturing lead time (invertir: menos tiempo = mejor)
    df['mfg_lead_time_norm'] = 1 - ((df['Manufacturing lead time'] - df['Manufacturing lead time'].min()) / 
                                   (df['Manufacturing lead time'].max() - df['Manufacturing lead time'].min() + 0.001))
    
    # Crear índice compuesto de eficiencia
    df['supply_chain_efficiency_index'] = (
        df['lead_time_norm'] * 0.4 +
        df['shipping_time_norm'] * 0.3 +
        df['mfg_lead_time_norm'] * 0.3
    )
    
    print("✅ supply_chain_efficiency_index: Índice compuesto de eficiencia temporal")
    print(f"   Media: {df['supply_chain_efficiency_index'].mean():.3f}")
    print(f"   Interpretación: Más alto = supply chain más eficiente en tiempos")
    new_supply_features.append('supply_chain_efficiency_index')
except Exception as e:
    print(f"⚠️ Error creando supply_chain_efficiency_index: {e}")

# Feature 2: Cost-to-Revenue Ratio
try:
    # Calcular ratio de costos totales vs revenue generado
    df['total_costs'] = (df['Shipping costs'] + df['Manufacturing costs'] + df['Costs'])
    df['cost_to_revenue_ratio'] = df['total_costs'] / (df['Revenue generated'] + 0.001)  # Evitar división por 0
    
    print("\n✅ cost_to_revenue_ratio: Ratio de eficiencia de costos")
    print(f"   Media: {df['cost_to_revenue_ratio'].mean():.3f}")
    print(f"   Interpretación: Más bajo = mejor eficiencia de costos")
    new_supply_features.append('cost_to_revenue_ratio')
except Exception as e:
    print(f"⚠️ Error creando cost_to_revenue_ratio: {e}")

# Feature 3: Quality Performance Score
try:
    # Crear score de calidad basado en defect rates e inspection results
    
    # Normalizar defect rates (invertir: menos defectos = mejor)
    df['defect_rate_norm'] = 1 - ((df['Defect rates'] - df['Defect rates'].min()) / 
                                  (df['Defect rates'].max() - df['Defect rates'].min() + 0.001))
    
    # Score por inspection results
    inspection_scores = {'Pass': 1.0, 'Pending': 0.5, 'Fail': 0.0}
    df['inspection_score'] = df['Inspection results'].map(inspection_scores)
    
    # Score compuesto de calidad
    df['quality_performance_score'] = (
        df['defect_rate_norm'] * 0.6 +
        df['inspection_score'] * 0.4
    )
    
    print("\n✅ quality_performance_score: Score de performance de calidad")
    print(f"   Media: {df['quality_performance_score'].mean():.3f}")
    print(f"   Interpretación: Más alto = mejor calidad del producto")
    new_supply_features.append('quality_performance_score')
except Exception as e:
    print(f"⚠️ Error creando quality_performance_score: {e}")

# Feature 4: Logistics Optimization Index
try:
    # Combinar métricas de shipping y transportation para optimización logística
    
    # Crear score por transportation mode (asumiendo Air=más rápido pero caro, Sea=económico pero lento)
    transport_efficiency = {'Air': 0.9, 'Road': 0.7, 'Rail': 0.6, 'Sea': 0.4}
    df['transport_efficiency_score'] = df['Transportation modes'].map(transport_efficiency)
    
    # Normalizar shipping costs (invertir: menos costo = mejor)
    df['shipping_cost_norm'] = 1 - ((df['Shipping costs'] - df['Shipping costs'].min()) / 
                                    (df['Shipping costs'].max() - df['Shipping costs'].min() + 0.001))
    
    # Índice de optimización logística
    df['logistics_optimization_index'] = (
        df['transport_efficiency_score'] * 0.4 +
        df['shipping_cost_norm'] * 0.3 +
        df['shipping_time_norm'] * 0.3
    )
    
    print("\n✅ logistics_optimization_index: Índice de optimización logística")
    print(f"   Media: {df['logistics_optimization_index'].mean():.3f}")
    print(f"   Interpretación: Más alto = logística más optimizada")
    new_supply_features.append('logistics_optimization_index')
except Exception as e:
    print(f"⚠️ Error creando logistics_optimization_index: {e}")

# Feature 5: Supplier Risk Index
try:
    # Crear índice de riesgo basado en concentración de supplier
    supplier_counts = df['Supplier name'].value_counts()
    df['supplier_dependency'] = df['Supplier name'].map(supplier_counts)
    
    # Normalizar (más productos de un supplier = mayor riesgo de concentración)

    df['supplier_risk_index'] = df['supplier_dependency'] / max_dependency
    
    print("\n✅ supplier_risk_index: Índice de riesgo de concentración de supplier")
    print(f"   Media: {df['supplier_risk_index'].mean():.3f}")
    print(f"   Interpretación: Más alto = mayor dependencia/riesgo de concentración")
    new_supply_features.append('supplier_risk_index')
except Exception as e:
    print(f"⚠️ Error creando supplier_risk_index: {e}")

print("\n🔧 FASE 4: ENGINEER - Creando features de supply chain con valor operacional")

new_supply_features = []

# Feature 1: Supply Chain Efficiency Index
try:
    # Combinar métricas de tiempo y costo para crear índice de eficiencia
    # Normalizar cada métrica (0-1) donde más alto = mejor
    
    # Normalizar lead times (invertir: menos tiempo = mejor)
    df['lead_time_norm'] = 1 - ((df['Lead times'] - df['Lead times'].min()) / 
                                (df['Lead times'].max() - df['Lead times'].min() + 0.001))
    
    # Normalizar shipping times (invertir: menos tiempo = mejor)
    df['shipping_time_norm'] = 1 - ((df['Shipping times'] - df['Shipping times'].min()) / 
                                    (df['Shipping times'].max() - df['Shipping times'].min() + 0.001))
    
    # Normalizar manufacturing lead time (invertir: menos tiempo = mejor)
    df['mfg_lead_time_norm'] = 1 - ((df['Manufacturing lead time'] - df['Manufacturing lead time'].min()) / 
                                   (df['Manufacturing lead time'].max() - df['Manufacturing lead time'].min() + 0.001))
    
    # Crear índice compuesto de eficiencia
    df['supply_chain_efficiency_index'] = (
        df['lead_time_norm'] * 0.4 +
        df['shipping_time_norm'] * 0.3 +
        df['mfg_lead_time_norm'] * 0.3
    )
    
    print("✅ supply_chain_efficiency_index: Índice compuesto de eficiencia temporal")
    print(f"   Media: {df['supply_chain_efficiency_index'].mean():.3f}")
    print(f"   Interpretación: Más alto = supply chain más eficiente en tiempos")
    new_supply_features.append('supply_chain_efficiency_index')
except Exception as e:
    print(f"⚠️ Error creando supply_chain_efficiency_index: {e}")

# Feature 2: Cost-to-Revenue Ratio
try:
    # Calcular ratio de costos totales vs revenue generado
    df['total_costs'] = (df['Shipping costs'] + df['Manufacturing costs'] + df['Costs'])
    df['cost_to_revenue_ratio'] = df['total_costs'] / (df['Revenue generated'] + 0.001)  # Evitar división por 0
    
    print("\n✅ cost_to_revenue_ratio: Ratio de eficiencia de costos")
    print(f"   Media: {df['cost_to_revenue_ratio'].mean():.3f}")
    print(f"   Interpretación: Más bajo = mejor eficiencia de costos")
    new_supply_features.append('cost_to_revenue_ratio')
except Exception as e:
    print(f"⚠️ Error creando cost_to_revenue_ratio: {e}")

# Feature 3: Quality Performance Score
try:
    # Crear score de calidad basado en defect rates e inspection results
    
    # Normalizar defect rates (invertir: menos defectos = mejor)
    df['defect_rate_norm'] = 1 - ((df['Defect rates'] - df['Defect rates'].min()) / 
                                  (df['Defect rates'].max() - df['Defect rates'].min() + 0.001))
    
    # Score por inspection results
    inspection_scores = {'Pass': 1.0, 'Pending': 0.5, 'Fail': 0.0}
    df['inspection_score'] = df['Inspection results'].map(inspection_scores)
    
    # Score compuesto de calidad
    df['quality_performance_score'] = (
        df['defect_rate_norm'] * 0.6 +
        df['inspection_score'] * 0.4
    )
    
    print("\n✅ quality_performance_score: Score de performance de calidad")
    print(f"   Media: {df['quality_performance_score'].mean():.3f}")
    print(f"   Interpretación: Más alto = mejor calidad del producto")
    new_supply_features.append('quality_performance_score')
except Exception as e:
    print(f"⚠️ Error creando quality_performance_score: {e}")

# Feature 4: Logistics Optimization Index
try:
    # Combinar métricas de shipping y transportation para optimización logística
    
    # Crear score por transportation mode (asumiendo Air=más rápido pero caro, Sea=económico pero lento)
    transport_efficiency = {'Air': 0.9, 'Road': 0.7, 'Rail': 0.6, 'Sea': 0.4}
    df['transport_efficiency_score'] = df['Transportation modes'].map(transport_efficiency)
    
    # Normalizar shipping costs (invertir: menos costo = mejor)
    df['shipping_cost_norm'] = 1 - ((df['Shipping costs'] - df['Shipping costs'].min()) / 
                                    (df['Shipping costs'].max() - df['Shipping costs'].min() + 0.001))
    
    # Índice de optimización logística
    df['logistics_optimization_index'] = (
        df['transport_efficiency_score'] * 0.4 +
        df['shipping_cost_norm'] * 0.3 +
        df['shipping_time_norm'] * 0.3
    )
    
    print("\n✅ logistics_optimization_index: Índice de optimización logística")
    print(f"   Media: {df['logistics_optimization_index'].mean():.3f}")
    print(f"   Interpretación: Más alto = logística más optimizada")
    new_supply_features.append('logistics_optimization_index')
except Exception as e:
    print(f"⚠️ Error creando logistics_optimization_index: {e}")

# Feature 5: Supplier Risk Index
try:
    # Crear índice de riesgo basado en concentración de supplier
    supplier_counts = df['Supplier name'].value_counts()
    df['supplier_dependency'] = df['Supplier name'].map(supplier_counts)
    
    # Normalizar (más productos de un supplier = mayor riesgo de concentración)
    max_dependency = df['supplier_dependency'].max()
    df['supplier_risk_index'] = df['supplier_dependency'] / max_dependency
    
    print("\n✅ supplier_risk_index: Índice de riesgo de concentración de supplier")
    print(f"   Media: {df['supplier_risk_index'].mean():.3f}")
    print(f"   Interpretación: Más alto = mayor dependencia/riesgo de concentración")
    new_supply_features.append('supplier_risk_index')
except Exception as e:
    print(f"⚠️ Error creando supplier_risk_index: {e}")

# Feature 6: Production Efficiency Ratio
try:
    # Relacionar production volumes con manufacturing lead time y costs
    df['production_efficiency_ratio'] = (
        df['Production volumes'] / 
        (df['Manufacturing lead time'] * df['Manufacturing costs'] + 0.001)
    )
    
    print("\n✅ production_efficiency_ratio: Ratio de eficiencia de producción")
    print(f"   Media: {df['production_efficiency_ratio'].mean():.3f}")
    print(f"   Interpretación: Más alto = mayor eficiencia de producción")
    new_supply_features.append('production_efficiency_ratio')
except Exception as e:
    print(f"⚠️ Error creando production_efficiency_ratio: {e}")
