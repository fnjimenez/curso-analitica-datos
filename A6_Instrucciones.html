print("\n⚙️ FASE 3: REUTILIZAR - Extrayendo valor de variables 'inútiles'")

insights_found = []

# 💎 INSIGHT 1: Employee Number patterns (cohortes empresariales ocultas)
print("\n💎 INSIGHT 1: Patrones Ocultos en Employee Numbers")
try:
    df['employee_cohort'] = pd.cut(df['employeenumber'], 
                                  bins=[0, 500, 1000, 1500, 2000], 
                                  labels=['Founders', 'Growth', 'Scale', 'Recent'])
    
    cohort_attrition = df.groupby('employee_cohort')['attrition'].apply(
        lambda x: (x == 'Yes').mean() * 100
    )
    
    print("🏢 Tasa de attrición por cohorte de empleado:")
    for cohort, rate in cohort_attrition.items():
        print(f"   • {cohort} Era: {rate:.1f}%")
    
    insights_found.append("Employee Number revela cohortes con diferentes comportamientos")
except Exception as e:
    print(f"⚠️ Error en análisis de cohortes: {e}")

# 💎 INSIGHT 2: StandardHours como baseline empresarial
print("\n💎 INSIGHT 2: StandardHours como Referencia Empresarial")
try:
    standard_hours = df['standardhours'].iloc[0]
    overtime_employees = df[df['overtime'] == 'Yes'].shape[0] if 'overtime' in df.columns else 0
    
    print(f"🕒 Horas estándar empresa: {standard_hours}")
    print(f"👥 Empleados con overtime: {overtime_employees}")
    print(f"💡 Insight: StandardHours = {standard_hours} es el baseline para calcular carga laboral real")
    
    insights_found.append("StandardHours revela política de horas base de la empresa")
except Exception as e:
    print(f"⚠️ Error en análisis StandardHours: {e}")

# 💎 INSIGHT 3: Over18 revela políticas de contratación

    print(f"💡 Insight: Política empresarial estricta - solo empleados mayores de 18 años")
    print(f"🔍 Esto indica compliance con regulaciones laborales internacionales")

Python - Fase 4
# ===================================================================
# ⚙️ FASE 4: ENGINEER - Crear Features Derivadas con Valor
# ===================================================================

print("\n🔧 FASE 4: ENGINEER - Creando features derivadas valiosas")

new_features_created = []

# Feature 1: Career Progression Rate (velocidad de ascenso)
try:
    df['career_progression_rate'] = df['joblevel'] / (df['totalworkingyears'] + 1)
    print("✅ career_progression_rate: Velocidad de progresión profesional")
    print(f"   Media: {df['career_progression_rate'].mean():.3f}")
    print(f"   Interpretación: Más alto = ascenso más rápido")
    new_features_created.append('career_progression_rate')
except Exception as e:
    print(f"⚠️ Error creando career_progression_rate: {e}")

# Feature 2: Stability Index (índice de estabilidad laboral)
try:
    df['stability_index'] = df['yearsatcompany'] / df['numcompaniesworked'].clip(lower=1)
    print("\n✅ stability_index: Índice de estabilidad laboral")
    print(f"   Media: {df['stability_index'].mean():.3f}")
    print(f"   Interpretación: Más alto = mayor lealtad/estabilidad")
    new_features_created.append('stability_index')
except Exception as e:
    print(f"⚠️ Error creando stability_index: {e}")

# Feature 3: Promotion Gap (brecha de promoción)
try:
    df['promotion_gap'] = df['yearsincurrentrole'] - df['yearssincelastpromotion']
    print("\n✅ promotion_gap: Brecha entre rol actual y última promoción")
    print(f"   Media: {df['promotion_gap'].mean():.3f}")
    print(f"   Interpretación: Positivo = tiempo desde promoción > tiempo en rol")
    new_features_created.append('promotion_gap')
except Exception as e:
    print(f"⚠️ Error creando promotion_gap: {e}")

# Feature 4: Satisfaction Consistency (consistencia en satisfacción)
try:
    satisfaction_cols = ['jobsatisfaction', 'environmentsatisfaction', 'relationshipsatisfaction']
    df['satisfaction_variance'] = df[satisfaction_cols].var(axis=1)
    print("\n✅ satisfaction_variance: Consistencia en satisfacción")
    print(f"   Media: {df['satisfaction_variance'].mean():.3f}")
    print(f"   Interpretación: Bajo = satisfacción consistente, Alto = mixta")
    new_features_created.append('satisfaction_variance')
except Exception as e:
    print(f"⚠️ Error creando satisfaction_variance: {e}")

# Feature 5: Income-Education Ratio (compensación vs educación)
try:
    education_mapping = {
        'Below College': 1, 'College': 2, 'Bachelor': 3, 'Master': 4, 'Doctor': 5
    }
    df['education_numeric'] = df['educationfield'].map(education_mapping).fillna(2)
    df['income_education_ratio'] = df['monthlyincome'] / (df['education_numeric'] * 1000)
    print("\n✅ income_education_ratio: Compensación vs nivel educativo")
    print(f"   Media: {df['income_education_ratio'].mean():.3f}")
    print(f"   Interpretación: Más alto = mejor compensado para su educación")
    new_features_created.append('income_education_ratio')
except Exception as e:
    print(f"⚠️ Error creando income_education_ratio: {e}")

print(f"\n✅ FASE 4 COMPLETADA - {len(new_features_created)} nuevas features creadas")
print(f"📊 Features: {new_features_created}")

# ===== SETUP ESPECÍFICO PARA VALIDACIÓN ESTADÍSTICA HR =====
library(tidyverse)    # Manipulación de datos y visualización
library(corrplot)     # Matriz de correlación visualizada
library(FactoMineR)   # PCA y análisis factorial
library(factoextra)   # Visualización PCA
library(randomForest) # Feature importance
library(survival)     # Análisis de supervivencia
library(survminer)    # Visualización survival

# Cargar dataset enriquecido de Python
hr_enriched <- read_csv("hr_value_extracted_python.csv")

cat("📊 Dataset cargado:", nrow(hr_enriched), "filas x", ncol(hr_enriched), "columnas\n")

# ===== 1. VALIDACIÓN ESTADÍSTICA DE INSIGHTS =====

# Test chi-cuadrado para Employee_Cohort vs Attrition
cohort_table <- table(hr_enriched$employee_cohort, hr_enriched$attrition)
cohort_test <- chisq.test(cohort_table)

cat("\n=== 🔬 CHI-SQUARE TEST: Employee Cohort vs Attrition ===\n")
print(cohort_test)

# Effect size (Cramér's V) - Mide la fuerza de la asociación
n <- sum(cohort_table)
chi_sq <- cohort_test$statistic
min_dim <- min(dim(cohort_table)) - 1
cramers_v <- sqrt(chi_sq / (n * min_dim))

cat("📈 Cramér's V (effect size):", round(cramers_v, 3), "\n")
cat("💡 Interpretación:", 
    ifelse(cramers_v < 0.1, "Efecto débil",
           ifelse(cramers_v < 0.3, "Efecto moderado", "Efecto fuerte")), "\n")

# ===== 2. ANÁLISIS FACTORIAL DE SATISFACCIONES =====
satisfaction_vars <- hr_enriched %>% 
  select(jobsatisfaction, environmentsatisfaction, 
         relationshipsatisfaction, worklifebalance) %>%
  na.omit()

# PCA de satisfacciones para encontrar dimensiones latentes
pca_satisfaction <- PCA(satisfaction_vars, graph = FALSE)

cat("\n=== 📊 PCA SATISFACCIÓN: Varianza Explicada ===\n")
print(round(pca_satisfaction$eig, 2))

# Visualización PCA con colores por contribución
pca_plot <- fviz_pca_var(pca_satisfaction, 
                             col.var = "contrib",
                             gradient.cols = c("#00AFBB", "#E7B800", "#FC4E07"),
                             repel = TRUE,
                             title = "💡 PCA: Dimensiones de Satisfacción")
print(pca_plot)

# ===== 3. SURVIVAL ANALYSIS (Tiempo hasta Attrición) =====
# Crear variable de tiempo usando yearsatcompany como proxy
hr_survival <- hr_enriched %>%
  mutate(
    event = ifelse(attrition == "Yes", 1, 0),  # 1 = evento ocurrió
    time = yearsatcompany                      # Tiempo observado
  ) %>%
  filter(!is.na(employee_cohort))

# Kaplan-Meier estimator por employee_cohort
km_fit <- survfit(Surv(time, event) ~ employee_cohort, 
                    data = hr_survival)

cat("\n=== ⏱️ SURVIVAL ANALYSIS: Tiempo hasta Attrición por Cohort ===\n")
print(km_fit)

# Visualizar curvas de supervivencia
survival_plot <- ggsurvplot(km_fit, 
                               data = hr_survival,
                               pval = TRUE,          # Mostrar p-value
                               conf.int = TRUE,     # Intervalos de confianza
                               risk.table = TRUE,   # Tabla de riesgo
                               title = "📈 Curvas de Supervivencia por Cohorte de Empleado",
                               xlab = "Años en la Empresa",
                               ylab = "Probabilidad de Retención")
print(survival_plot)

# ===== 4. RANDOM FOREST PARA FEATURE IMPORTANCE =====
# Preparar datos para modelo (solo variables numéricas y categóricas válidas)
hr_model <- hr_enriched %>%
  select(-employeenumber) %>%                   # Remover ID
  mutate(attrition = as.factor(attrition)) %>%  # Convertir a factor
  na.omit()                                       # Eliminar NAs

# Ejecutar Random Forest
set.seed(42)  # Para reproducibilidad
rf_model <- randomForest(attrition ~ ., 
                           data = hr_model, 
                           importance = TRUE,
                           ntree = 500)

# Extraer importancia de variables
importance_df <- data.frame(
  Variable = rownames(rf_model$importance),
  Importance = rf_model$importance[, "MeanDecreaseGini"]
) %>%
  arrange(desc(Importance)) %>%
  slice_head(n = 10)

cat("\n=== 🏆 TOP 10 VARIABLES MÁS IMPORTANTES ===\n")
print(importance_df)

# ===== 5. VISUALIZACIONES AVANZADAS CON GGPLOT2 =====

# Gráfico de insights contraintuitivos
counterintuitive_plot <- hr_enriched %>%
  mutate(
    satisfaction_level = case_when(
      jobsatisfaction >= 3 & environmentsatisfaction >= 3 ~ "High Satisfaction",
      jobsatisfaction <= 2 & environmentsatisfaction <= 2 ~ "Low Satisfaction",
      TRUE ~ "Mixed Satisfaction"
    )
  ) %>%
  count(satisfaction_level, attrition) %>%
  group_by(satisfaction_level) %>%
  mutate(prop = n / sum(n)) %>%
  filter(attrition == "Yes") %>%
  ggplot(aes(x = reorder(satisfaction_level, prop), 
             y = prop, 
             fill = satisfaction_level)) +
  geom_col(alpha = 0.8) +
  geom_text(aes(label = paste0(round(prop * 100, 1), "%")), 
            vjust = -0.5, size = 4, fontface = "bold") +
  scale_y_continuous(labels = scales::percent) +
  labs(
    title = "🤔 INSIGHT CONTRAINTUITIVO: Attrición por Nivel de Satisfacción",
    subtitle = "¿Por qué empleados con alta satisfacción también se van?",
    x = "Nivel de Satisfacción",
    y = "Tasa de Attrición",
    caption = "Fuente: HR Attrition Dataset - Análisis VDRER"
  ) +
  theme_minimal() +
  theme(legend.position = "none")

print(counterintuitive_plot)

# ===== 6. EXPORT FINAL PARA POWER BI =====
write_csv(hr_enriched, "hr_final_statistical_validated.csv")
