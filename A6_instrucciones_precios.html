        print(f"📦 Análisis de disponibilidad de stock:")

        print(f"   • Productos disponibles: {available_count:,} de {total_products:,}")

        insights_found.append("Stock Availability alta revela estrategia de gestión de inventario eficiente")

Python - Fase 4
# ===================================================================
# ⚙️ FASE 4: ENGINEER - Crear Features Derivadas con Valor Comercial
# ===================================================================

print("\n🔧 FASE 4: ENGINEER - Creando features derivadas comerciales valiosas")

new_features_created = []

# Feature 1: Discount Percentage (porcentaje de descuento)
try:
    df['discount_percentage'] = ((df['mrp'] - df['price']) / df['mrp'] * 100).clip(0, 100)
    print("✅ discount_percentage: Porcentaje de descuento aplicado")
    print(f"   Media: {df['discount_percentage'].mean():.1f}%")
    print(f"   Interpretación: Más alto = mayor descuento aplicado")
    new_features_created.append('discount_percentage')
except Exception as e:
    print(f"⚠️ Error creando discount_percentage: {e}")

# Feature 2: Price Premium Index (índice de premium vs categoría)
try:
    category_avg_price = df.groupby('bb_category')['price'].mean()
    df['price_premium_index'] = df.apply(
        lambda row: row['price'] / category_avg_price[row['bb_category']], axis=1
    )
    print("\n✅ price_premium_index: Índice de premium vs categoría")
    print(f"   Media: {df['price_premium_index'].mean():.3f}")
    print(f"   Interpretación: >1 = precio premium, <1 = precio económico")
    new_features_created.append('price_premium_index')
except Exception as e:
    print(f"⚠️ Error creando price_premium_index: {e}")

# Feature 3: Brand Market Position (posición de marca en el mercado)
try:
    brand_avg_price = df.groupby('brand')['price'].mean()
    overall_avg_price = df['price'].mean()
    
    brand_position_map = {}
    for brand, avg_price in brand_avg_price.items():
        if avg_price > overall_avg_price * 1.5:
            brand_position_map[brand] = 'Premium'
        elif avg_price > overall_avg_price * 0.8:
            brand_position_map[brand] = 'Mid-Market'
        else:
            brand_position_map[brand] = 'Economy'
    
    df['brand_market_position'] = df['brand'].map(brand_position_map)
    print("\n✅ brand_market_position: Posición de marca en mercado")
    position_dist = df['brand_market_position'].value_counts()
    print(f"   Distribución: {position_dist.to_dict()}")
    new_features_created.append('brand_market_position')
except Exception as e:
    print(f"⚠️ Error creando brand_market_position: {e}")

# Feature 4: Product Value Score (score de valor del producto)
try:
    # Normalizar variables para crear score compuesto
    discount_norm = df['discount_percentage'] / 100  # 0-1
    price_affordability = 1 - (df['price'] / df['price'].max())  # 0-1, más alto = más accesible
    
    df['product_value_score'] = (discount_norm * 0.6 + price_affordability * 0.4)
    print("\n✅ product_value_score: Score de valor del producto")
    print(f"   Media: {df['product_value_score'].mean():.3f}")
    print(f"   Interpretación: Más alto = mejor valor (descuento + accesibilidad)")
    new_features_created.append('product_value_score')
except Exception as e:
    print(f"⚠️ Error creando product_value_score: {e}")

# Feature 5: Category Competition Level (nivel de competencia por categoría)
try:
    category_brand_count = df.groupby('bb_category')['brand'].nunique()
    category_product_count = df.groupby('bb_category').size()
    
    df['category_competition_level'] = df['bb_category'].map(category_brand_count)
    print("\n✅ category_competition_level: Nivel de competencia por categoría")
    print(f"   Media: {df['category_competition_level'].mean():.1f} marcas por categoría")
    print(f"   Interpretación: Más alto = mayor competencia en la categoría")
    new_features_created.append('category_competition_level')
except Exception as e:
    print(f"⚠️ Error creando category_competition_level: {e}")

print(f"\n✅ FASE 4 COMPLETADA - {len(new_features_created)} nuevas features comerciales creadas")
print(f"📊 Features: {new_features_created}")

# ===== SETUP ESPECÍFICO PARA VALIDACIÓN ESTADÍSTICA ANÁLISIS DE PRECIOS =====
library(tidyverse)    # Manipulación de datos y visualización
library(corrplot)     # Matriz de correlación visualizada
library(FactoMineR)   # PCA y análisis factorial
library(factoextra)   # Visualización PCA
library(randomForest) # Feature importance
library(car)          # ANOVA avanzado
library(broom)        # Tidy statistical output

# Cargar dataset enriquecido de Python
pricing_enriched <- read_csv("analiz_precios_value_extracted_python.csv")

cat("🛒 Dataset comercial cargado:", nrow(pricing_enriched), "productos x", ncol(pricing_enriched), "variables\n")

# ===== 1. VALIDACIÓN ESTADÍSTICA DE INSIGHTS COMERCIALES =====

# Test chi-cuadrado para Brand_Market_Position vs Category
brand_category_table <- table(pricing_enriched$brand_market_position, pricing_enriched$bb_category)
brand_test <- chisq.test(brand_category_table)

cat("\n=== 🔬 CHI-SQUARE TEST: Posición de Marca vs Categoría ===\n")
print(brand_test)

# Effect size (Cramér's V) - Mide la fuerza de la asociación comercial
n <- sum(brand_category_table)
chi_sq <- brand_test$statistic
min_dim <- min(dim(brand_category_table)) - 1
cramers_v <- sqrt(chi_sq / (n * min_dim))

cat("📈 Cramér's V (effect size comercial):", round(cramers_v, 3), "\n")
cat("💡 Interpretación comercial:", 
    ifelse(cramers_v < 0.1, "Independencia entre marca y categoría",
           ifelse(cramers_v < 0.3, "Asociación moderada marca-categoría", 
                  "Fuerte asociación estratégica marca-categoría")), "\n")

# ===== 2. ANÁLISIS FACTORIAL DE VARIABLES DE PRICING =====
pricing_vars <- pricing_enriched %>% 
  select(price, mrp, discount_percentage, 
         price_premium_index, product_value_score) %>%
  na.omit()

# PCA de variables de pricing para encontrar dimensiones latentes
pca_pricing <- PCA(pricing_vars, graph = FALSE)

cat("\n=== 📊 PCA VARIABLES DE PRICING: Varianza Explicada ===\n")
print(round(pca_pricing$eig, 2))

# Visualización PCA con colores por contribución comercial
pca_plot <- fviz_pca_var(pca_pricing, 
                             col.var = "contrib",
                             gradient.cols = c("#228B22", "#FFD700", "#FF6347"),
                             repel = TRUE,
                             title = "💡 PCA: Dimensiones de Estrategia de Precios")
print(pca_plot)

# ===== 3. ANÁLISIS DE VARIANZA (ANOVA) PARA ESTRATEGIAS DE PRICING =====
# ANOVA para precio por posición de marca
anova_brand_price <- aov(price ~ brand_market_position, data = pricing_enriched)

cat("\n=== 📊 ANOVA: Precio por Posición de Marca ===\n")
print(summary(anova_brand_price))

# Post-hoc tests para identificar diferencias específicas
tukey_brand <- TukeyHSD(anova_brand_price)
print(tukey_brand)

# Preparar datos para modelo (predecir categoría de precio)
pricing_model_data <- pricing_enriched %>%
  mutate(
    price_category = case_when(
      price <= quantile(price, 0.33, na.rm = TRUE) ~ "Low",
      price <= quantile(price, 0.67, na.rm = TRUE) ~ "Medium",
      TRUE ~ "High"
    )
  ) %>%
  select(discount_percentage, price_premium_index, product_value_score,
         category_competition_level, price_category) %>%
  na.omit() %>%
  mutate(price_category = as.factor(price_category))

# Ejecutar Random Forest para análisis comercial
set.seed(42)  # Para reproducibilidad

                                   data = pricing_model_data, 
                                   importance = TRUE,
                                   ntree = 500)

# Extraer importancia de variables comerciales
importance_pricing_df <- data.frame(
  Variable = rownames(rf_pricing_model$importance),
  Importance = rf_pricing_model$importance[, "MeanDecreaseGini"]
) %>%
  arrange(desc(Importance))
