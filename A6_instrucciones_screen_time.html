import pandas as pd

df = pd.read_csv('analiz_screen_time.csv')
print(f"✅ Dataset cargado: {df.shape[0]:,} filas × {df.shape[1]} columnas")

# Overview del comportamiento digital
print(f"\n📱 OVERVIEW DIGITAL BEHAVIOR:")
print(f"👥 Total Users: {df['User ID'].nunique():,}")
print(f"📱 Device Models: {df['Device Model'].nunique():,}")
print(f"💻 Operating Systems: {df['Operating System'].value_counts().to_dict()}")
print(f"🎂 Age Range: {df['Age'].min()}-{df['Age'].max()} años")
print(f"⚡ Avg Screen Time: {df['Screen On Time (hours/day)'].mean():.1f} hrs/day")
print(f"📊 Avg App Usage: {df['App Usage Time (min/day)'].mean():.0f} min/day")

# Behavior classes overview
print(f"\n🎯 USER BEHAVIOR CLASSES DISTRIBUTION:")
behavior_stats = df['User Behavior Class'].value_counts().sort_index()
for class_num, count in behavior_stats.items():
    pct = count / len(df) * 100
    print(f"   Class {class_num}: {count} users ({pct:.1f}%)")

# ===== FASE 2: DESCUBRIR =====
print(f"\n🔍 FASE 2: DESCUBRIR - Variables 'inútiles' digitales")

insights_found = []

# User ID patterns analysis
print(f"\n📊 ANÁLISIS USER ID:")
print(f"Rango: {df['User ID'].min()} - {df['User ID'].max()}")
user_id_gaps = df['User ID'].diff().dropna()
print(f"Gaps en secuencia: mean={user_id_gaps.mean():.1f}, std={user_id_gaps.std():.1f}")

# Check if User ID correlates with behavior patterns (early adopters vs late adopters)
correlation_userid_usage = df['User ID'].corr(df['App Usage Time (min/day)'])
print(f"Correlación User ID vs App Usage: {correlation_userid_usage:.3f}")

# Device Model analysis
print(f"\n📊 ANÁLISIS DEVICE MODEL:")
print(f"Ejemplo: {df['Device Model'].iloc[0]}")
device_brands = df['Device Model'].str.split().str[0].value_counts()
print(f"Top device brands: {device_brands.head().to_dict()}")

# ===== FASE 3: REUTILIZAR =====
print(f"\n♻️ FASE 3: REUTILIZAR - Extrayendo valor de códigos digitales")

# User adoption cohorts (basado en User ID como proxy de signup timing)
df['user_cohort'] = pd.qcut(df['User ID'], q=4, labels=['EarlyAdopters', 'Growth', 'Expansion', 'Recent'])

# Device ecosystem extraction
df['device_brand'] = df['Device Model'].str.split().str[0]
df['device_generation'] = df['Device Model'].str.extract(r'(\d+)').fillna('0').astype(int)

# OS ecosystem preferences
os_usage_patterns = df.groupby('Operating System').agg({
    'App Usage Time (min/day)': 'mean',
    'Screen On Time (hours/day)': 'mean',
    'Battery Drain (mAh/day)': 'mean'
}).round(2)

print(f"\n📱 OS ECOSYSTEM PATTERNS:")
for os in os_usage_patterns.index:
    app_time = os_usage_patterns.loc[os, 'App Usage Time (min/day)']
    screen_time = os_usage_patterns.loc[os, 'Screen On Time (hours/day)']
    battery = os_usage_patterns.loc[os, 'Battery Drain (mAh/day)']
    print(f"   {os}: {app_time:.0f}min apps, {screen_time:.1f}h screen, {battery:.0f}mAh battery")

# User Behavior Class deeper analysis
behavior_analysis = df.groupby('User Behavior Class').agg({
    'App Usage Time (min/day)': ['mean', 'std'],
    'Screen On Time (hours/day)': ['mean', 'std'],
    'Number of Apps Installed': 'mean'
}).round(2)

print(f"\n🎯 BEHAVIOR CLASS DEEP ANALYSIS:")
for class_num in sorted(df['User Behavior Class'].unique()):
    if class_num in behavior_analysis.index:
        app_mean = behavior_analysis.loc[class_num, ('App Usage Time (min/day)', 'mean')]
        screen_mean = behavior_analysis.loc[class_num, ('Screen On Time (hours/day)', 'mean')]
        apps_count = behavior_analysis.loc[class_num, ('Number of Apps Installed', 'mean')]
        print(f"   Class {class_num}: {app_mean:.0f}min apps, {screen_mean:.1f}h screen, {apps_count:.0f} apps")

insights_found.extend([
    "User ID revela adoption cohorts con diferentes usage patterns",
    "Device Model estructura indica brand ecosystems y generations",
    "Operating System correlaciona con usage efficiency patterns",
    "User Behavior Class codifica sophisticated digital wellness segments"
])

# ===== FASE 4: ENGINEER =====
print(f"\n🔧 FASE 4: ENGINEER - Creando features de digital wellness")

new_features_created = []

# Digital Wellness Index
# Combinar múltiples factores de healthy digital usage
df['screen_time_hours'] = df['Screen On Time (hours/day)']
df['app_usage_hours'] = df['App Usage Time (min/day)'] / 60

# Normalize values for scoring (lower screen time = higher wellness)
max_screen = df['screen_time_hours'].max()
max_app = df['app_usage_hours'].max()

df['digital_wellness_index'] = (
    (1 - df['screen_time_hours'] / max_screen) * 0.4 +  # Less screen time = better
    (1 - df['app_usage_hours'] / max_app) * 0.3 +  # Less app time = better
    (df['Number of Apps Installed'] / df['Number of Apps Installed'].max()) * 0.3  # More apps = more choice
)

# Device Efficiency Score
# Battery life vs usage efficiency
df['device_efficiency_score'] = (
    df['Screen On Time (hours/day)'] + (df['App Usage Time (min/day)'] / 60)
) / (df['Battery Drain (mAh/day)'] / 1000)  # Hours of usage per unit of battery

# User Engagement Level
# Based on app usage intensity and diversity
df['usage_intensity'] = (
    df['App Usage Time (min/day)'] / 60 +  # Hours of app usage
    df['Screen On Time (hours/day)'] +  # Hours of screen time
    df['Data Usage (MB/day)'] / 1000  # GB of data usage
) / 3

df['engagement_level'] = pd.cut(df['usage_intensity'],
                               bins=[0, 2, 4, 6, float('inf')],
                               labels=['Light', 'Moderate', 'Heavy', 'Extreme'])

# Battery Optimization Index
# How efficiently users manage battery vs usage
median_battery = df['Battery Drain (mAh/day)'].median()
df['battery_optimization_index'] = np.where(
    df['Battery Drain (mAh/day)'] <= median_battery,
    df['device_efficiency_score'] * 1.2,  # Bonus for low battery usage
    df['device_efficiency_score'] * 0.8   # Penalty for high battery usage
)

# Age-Digital Natives Score
# How digital behavior aligns with age expectations
df['digital_native_score'] = np.where(
    df['Age'] <= 25,  # Digital natives
    (df['App Usage Time (min/day)'] / 400) * (df['Number of Apps Installed'] / 50),  # Expected high usage
    (200 / df['App Usage Time (min/day)']) * (20 / df['Number of Apps Installed'])  # Expected moderate usage
).clip(0, 2)

# Device Ecosystem Loyalty
device_brand_counts = df['device_brand'].value_counts()
df['brand_popularity'] = df['device_brand'].map(device_brand_counts)
df['ecosystem_preference'] = df.apply(lambda row: 
    'Apple' if 'iPhone' in str(row['Device Model']) or row['Operating System'] == 'iOS'
    else 'Google' if row['Operating System'] == 'Android' 
    else 'Other', axis=1)

new_features_created.extend([
    'digital_wellness_index', 'device_efficiency_score', 'engagement_level',
    'battery_optimization_index', 'digital_native_score', 'ecosystem_preference'
])

print(f"✅ Features digitales creadas: {len(new_features_created)}")

# Performance analysis por cohort
cohort_analysis = df.groupby('user_cohort').agg({
    'digital_wellness_index': 'mean',
    'device_efficiency_score': 'mean',
    'App Usage Time (min/day)': 'mean'
}).round(3)

print(f"\n📈 PERFORMANCE BY USER COHORT:")
for cohort in cohort_analysis.index:
    wellness = cohort_analysis.loc[cohort, 'digital_wellness_index']
    efficiency = cohort_analysis.loc[cohort, 'device_efficiency_score']
    app_time = cohort_analysis.loc[cohort, 'App Usage Time (min/day)']
    print(f"   {cohort}: Wellness={wellness:.3f}, Efficiency={efficiency:.3f}, AppTime={app_time:.0f}min")

# ===== FASE 5: REPORTAR =====
print(f"\n📋 FASE 5: REPORTAR - Exportando dataset digital enriquecido")

print(f"\n📊 REPORTE FINAL DIGITAL:")
print(f"   💎 Insights de digital behavior extraídos: {len(insights_found)}")
print(f"   📱 Features de mobile analytics creadas: {len(new_features_created)}")
print(f"   📈 Dimensiones finales: {df.shape}")

# Digital wellness distribution
wellness_categories = pd.cut(df['digital_wellness_index'], 
                           bins=[0, 0.3, 0.6, 0.8, 1.0],
                           labels=['Poor', 'Fair', 'Good', 'Excellent'])
wellness_dist = wellness_categories.value_counts()
print(f"\n🌟 DIGITAL WELLNESS DISTRIBUTION:")
for category in ['Poor', 'Fair', 'Good', 'Excellent']:
    if category in wellness_dist.index:
        count = wellness_dist[category]
        pct = count / len(df) * 100
        print(f"   {category}: {count:,} users ({pct:.1f}%)")

# Engagement level distribution
engagement_dist = df['engagement_level'].value_counts()
print(f"\n⚡ USER ENGAGEMENT DISTRIBUTION:")
for level in ['Light', 'Moderate', 'Heavy', 'Extreme']:
    if level in engagement_dist.index:
        count = engagement_dist[level]
        pct = count / len(df) * 100
        print(f"   {level}: {count:,} users ({pct:.1f}%)")

# Ecosystem preference analysis
ecosystem_dist = df['ecosystem_preference'].value_counts()
print(f"\n🏢 ECOSYSTEM PREFERENCE:")
for ecosystem in ecosystem_dist.index:

# Export enhanced dataset
try:

screen_data <- read_csv("screen_time_value_extracted_python.csv")

cat("📱 SCREEN TIME ANALYTICS STATISTICAL VALIDATION\n")
cat("===============================================\n")

# Verificar features VDRER
vdrer_features <- c("digital_wellness_index", "device_efficiency_score", 
                    "engagement_level", "user_cohort", "ecosystem_preference")

for(feature in vdrer_features) {
  if(feature %in% names(screen_data)) {
    cat("✅", feature, "- Type:", class(screen_data[[feature]])[1], "\n")
  } else {
    cat("❌", feature, "- NOT FOUND\n")
  }
}

# TEST 1: ANOVA - Operating System vs Digital Wellness
if("Operating System" %in% names(screen_data) && "digital_wellness_index" %in% names(screen_data)) {
  
  anova_data <- screen_data %>%
    filter(!is.na(`Operating System`), !is.na(digital_wellness_index), 
           is.finite(digital_wellness_index))
  
  if(nrow(anova_data) > 0) {
    anova_result <- aov(digital_wellness_index ~ `Operating System`, data = anova_data)
    anova_summary <- summary(anova_result)
    
    cat("\n📊 ANOVA - Digital Wellness por Operating System:\n")
    cat("F-statistic:", round(anova_summary[[1]][1,4], 3), "\n")
    cat("p-value:", format(anova_summary[[1]][1,5], scientific = TRUE), "\n")
    
    if(anova_summary[[1]][1,5] < 0.05) {
      cat("✅ Diferencias significativas entre OS ecosystems\n")
      
      # Effect sizes
      group_means <- anova_data %>%
        group_by(`Operating System`) %>%
        summarise(mean_wellness = mean(digital_wellness_index, na.rm = TRUE))
      
      cat("📈 Digital Wellness por OS:\n")
      print(group_means)
    }
  }
}

# TEST 2: User Cohort Digital Behavior Evolution
if("user_cohort" %in% names(screen_data)) {
  
  cohort_analysis <- screen_data %>%
    group_by(user_cohort) %>%
    summarise(
      avg_screen_time = mean(`Screen On Time (hours/day)`, na.rm = TRUE),
      avg_wellness = mean(digital_wellness_index, na.rm = TRUE),
      avg_efficiency = mean(device_efficiency_score, na.rm = TRUE),
      user_count = n()
    ) %>%
    arrange(user_cohort)
  
  cat("\n📊 USER COHORT EVOLUTION ANALYSIS:\n")
  print(cohort_analysis)

  if("digital_wellness_index" %in% names(screen_data)) {
    cohort_numeric <- as.numeric(factor(screen_data$user_cohort,

    cor_test <- cor.test(cohort_numeric, screen_data$digital_wellness_index, 
                        method = "spearman", use = "complete.obs")
    
    cat("\nSpearman correlation (Cohort vs Digital Wellness):", round(cor_test$estimate, 3), "\n")
    cat("p-value:", format(cor_test$p.value, scientific = TRUE), "\n")
    
    if(cor_test$p.value < 0.05) {
      cat("✅ Trend significativo en digital wellness por user cohort\n")
    }
  }

    select(digital_wellness_index, device_efficiency_score, 
           `App Usage Time (min/day)`, `Screen On Time (hours/day)`) %>%
    na.omit() %>%
    filter(across(everything(), is.finite))
  
  if(nrow(cluster_features) > 100) {
    # K-means clustering
    cluster_scaled <- scale(cluster_features)
    
    set.seed(42)
    kmeans_result <- kmeans(cluster_scaled, centers = 4, nstart = 25)
    
    cat("\n🎯 USER BEHAVIOR CLUSTERING VALIDATION:\n")
    cat("Clusters identificados:", max(kmeans_result$cluster), "\n")
    
    # Silhouette analysis
    if(require(cluster, quietly = TRUE)) {
      sil_analysis <- silhouette(kmeans_result$cluster, dist(cluster_scaled))
      avg_sil_width <- mean(sil_analysis[, 3])
      cat("Silhouette Width:", round(avg_sil_width, 3), "\n")
      
      if(avg_sil_width > 0.5) {
        cat("✅ Clustering robusto - User behavior groups bien definidos\n")
      } else if(avg_sil_width > 0.25) {
        cat("⚠️ Clustering moderado - User groups identificables\n")
      } else {
        cat("❌ Clustering débil\n")
      }
      
      # Caracterizar clusters
      cluster_features$cluster <- kmeans_result$cluster
      cluster_summary <- cluster_features %>%
        group_by(cluster) %>%
        summarise(
          users = n(),
          avg_wellness = round(mean(digital_wellness_index), 3),
          avg_efficiency = round(mean(device_efficiency_score), 3),
          avg_screen_time = round(mean(`Screen On Time (hours/day)`), 1)
        )
      
      cat("📊 Clusters caracterizados:\n")
      print(cluster_summary)
    }
  }
}

      rf_model <- randomForest(as.factor(`User Behavior Class`) ~ .,

      # Feature importance
      importance_scores <- importance(rf_model)
      importance_df <- data.frame(
        Feature = rownames(importance_scores),
        MeanDecreaseGini = importance_scores[, "MeanDecreaseGini"]
      ) %>%
        arrange(desc(MeanDecreaseGini))
      
      cat("\n🏆 FEATURE IMPORTANCE para User Behavior Class Prediction:\n")
      print(head(importance_df, 6))
      
      # Model performance
      cat("\nModel OOB Error Rate:", round(rf_model$err.rate[rf_model$ntree, "OOB"], 3), "\n")
      
      if(rf_model$err.rate[rf_model$ntree, "OOB"] < 0.4) {
        cat("✅ Modelo predictivo de behavior classification exitoso\n")
      }
    }
  }
