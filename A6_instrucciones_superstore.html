print("\n⚙️ FASE 3: REUTILIZAR - Extrayendo valor de IDs y códigos")

insights_found = []

# 💎 INSIGHT 1: Row ID como proxy temporal (si existe correlación)
print("\n💎 INSIGHT 1: Row ID y Timing de Negocios")
try:
    # Crear ventanas temporales basadas en Row ID
    df['row_id_quintile'] = pd.qcut(df['Row ID'], q=5, labels=['Q1', 'Q2', 'Q3', 'Q4', 'Q5'])
    
    # Analizar ventas por quintil de entrada
    sales_by_quintile = df.groupby('row_id_quintile')['Sales'].agg(['mean', 'count', 'sum'])
    
    print("📈 Ventas por quintil de Row ID (cronología de entrada):")
    for quintile in sales_by_quintile.index:
        avg_sales = sales_by_quintile.loc[quintile, 'mean']
        total_sales = sales_by_quintile.loc[quintile, 'sum']
        print(f"   {quintile}: Promedio ${avg_sales:.2f}, Total ${total_sales:,.2f}")
    
    insights_found.append("Row ID revela patterns de performance por período de entrada")
except Exception as e:
    print(f"⚠️ Error en análisis Row ID: {e}")

# 💎 INSIGHT 2: Order ID prefix patterns
print("\n💎 INSIGHT 2: Order ID - Patrones de Sistema/Región")
try:
    df['order_prefix'] = df['Order ID'].str[:2]
    
    # Analizar ventas por prefijo
    prefix_analysis = df.groupby('order_prefix').agg({
        'Sales': ['mean', 'sum', 'count'],
        'Region': lambda x: x.mode().iloc[0] if len(x.mode()) > 0 else 'Unknown'
    }).round(2)
    
    print("🎯 Top 5 prefijos de Order ID por ventas:")
    top_prefixes = df.groupby('order_prefix')['Sales'].sum().sort_values(ascending=False).head()
    for prefix, sales in top_prefixes.items():
        region = df[df['order_prefix'] == prefix]['Region'].mode().iloc[0]
        print(f"   Prefix '{prefix}': ${sales:,.2f} total, Región dominante: {region}")
    
    insights_found.append("Order ID prefixes correlacionan con regiones geográficas")
except Exception as e:
    print(f"⚠️ Error en análisis Order ID: {e}")

# 💎 INSIGHT 3: Customer ID encoding de segmentos
print("\n💎 INSIGHT 3: Customer ID - Segmentación Oculta")
try:
    df['customer_prefix'] = df['Customer ID'].str[:2]
    
    # Analizar segmentos por prefijo de Customer ID
    customer_analysis = df.groupby(['customer_prefix', 'Segment']).size().unstack(fill_value=0)
    
    print("👥 Distribución de segmentos por prefijo de Customer ID:")
    for prefix in customer_analysis.index[:5]:  # Top 5
        total = customer_analysis.loc[prefix].sum()
        if total > 0:
            dominant_segment = customer_analysis.loc[prefix].idxmax()
            percentage = customer_analysis.loc[prefix].max() / total * 100
            print(f"   Prefix '{prefix}': {total} customers, {percentage:.1f}% {dominant_segment}")
    
    insights_found.append("Customer ID prefixes reflejan estrategias de segmentación")
except Exception as e:
    print(f"⚠️ Error en análisis Customer ID: {e}")

# 💎 INSIGHT 4: Product ID hierarchy
print("\n💎 INSIGHT 4: Product ID - Jerarquía de Categorías")
try:
    df['product_prefix'] = df['Product ID'].str[:3]
    
    # Verificar si prefijos correlacionan con categorías
    product_category_map = df.groupby('product_prefix')['Category'].apply(
        lambda x: x.mode().iloc[0] if len(x.mode()) > 0 else 'Mixed'
    )
    
    print("📦 Product ID prefixes por categoría:")
    for category in df['Category'].unique():
        matching_prefixes = product_category_map[product_category_map == category]
        if len(matching_prefixes) > 0:
            print(f"   {category}: prefixes {list(matching_prefixes.index[:3])}")  # Top 3
    
    insights_found.append("Product ID structure codifica jerarquías de categorías")
except Exception as e:
    print(f"⚠️ Error en análisis Product ID: {e}")

print(f"\n✅ FASE 3 COMPLETADA - {len(insights_found)} insights valiosos extraídos")

# ⚙️ FASE 4: ENGINEER - Features de Retail Intelligence

print("\n🔧 FASE 4: ENGINEER - Creando features valiosas para retail")

new_features_created = []

# Feature 1: Customer Lifetime Value (CLV) simplificado
try:
    customer_metrics = df.groupby('Customer ID').agg({
        'Sales': ['sum', 'count', 'mean'],
        'Order Date': ['min', 'max']
    }).round(2)
    
    customer_metrics.columns = ['total_sales', 'order_count', 'avg_order_value', 'first_order', 'last_order']
    customer_metrics['customer_lifetime_days'] = (customer_metrics['last_order'] - customer_metrics['first_order']).dt.days + 1
    customer_metrics['clv_simple'] = customer_metrics['total_sales'] / customer_metrics['customer_lifetime_days'] * 365
    
    # Merge back to main dataframe
    df = df.merge(customer_metrics[['total_sales', 'order_count', 'clv_simple']], 
                  left_on='Customer ID', right_index=True, suffixes=('', '_customer'))
    
    print("✅ customer_lifetime_value: CLV anualizado por cliente")
    print(f"   Media CLV: ${df['clv_simple'].mean():.2f}")
    print(f"   Top CLV: ${df['clv_simple'].max():.2f}")
    new_features_created.append('clv_simple')
except Exception as e:
    print(f"⚠️ Error creando CLV: {e}")

# Feature 2: Product Performance Index
try:
    product_metrics = df.groupby('Product ID').agg({
        'Sales': ['sum', 'count', 'mean'],
        'Order ID': 'nunique'
    }).round(2)
    
    product_metrics.columns = ['product_total_sales', 'product_order_count', 'product_avg_sale', 'product_unique_orders']
    
    # Crear índice de performance (ventas * frecuencia)
    product_metrics['product_performance_index'] = (
        product_metrics['product_total_sales'] * 
        product_metrics['product_order_count'] / 
        product_metrics['product_total_sales'].max()
    )
    
    df = df.merge(product_metrics[['product_performance_index']], 
                  left_on='Product ID', right_index=True)
    
    print("✅ product_performance_index: Índice combinado ventas x frecuencia")
    print(f"   Media: {df['product_performance_index'].mean():.3f}")
    new_features_created.append('product_performance_index')
except Exception as e:
    print(f"⚠️ Error creando Product Performance Index: {e}")

# Feature 3: Geographic Revenue Density
try:
    # Revenue density por estado/región
    geo_metrics = df.groupby(['State', 'Region']).agg({
        'Sales': 'sum',
        'Customer ID': 'nunique'
    }).round(2)
    
    geo_metrics['revenue_per_customer'] = geo_metrics['Sales'] / geo_metrics['Customer ID']
    
    # Merge back 
    df = df.merge(geo_metrics[['revenue_per_customer']], 
                  left_on=['State', 'Region'], right_index=True)
    
    print("✅ geographic_revenue_density: Revenue por customer por estado")
    print(f"   Media: ${df['revenue_per_customer'].mean():.2f}")
    new_features_created.append('revenue_per_customer')
except Exception as e:
    print(f"⚠️ Error creando Geographic Revenue Density: {e}")

# Feature 4: Order Timing Intelligence
try:
    # Calcular días entre order y ship
    df['order_to_ship_days'] = (df['Ship Date'] - df['Order Date']).dt.days
    
    # Día de la semana y mes del año
    df['order_day_of_week'] = df['Order Date'].dt.day_name()
    df['order_month'] = df['Order Date'].dt.month
    df['order_quarter'] = df['Order Date'].dt.quarter
    
    # Scoring de eficiencia de shipping
    df['shipping_efficiency'] = np.where(df['order_to_ship_days'] <= 2, 'Fast',
                                np.where(df['order_to_ship_days'] <= 5, 'Normal', 'Slow'))
    
    print("✅ order_timing_features: Análisis temporal de órdenes")
    print(f"   Promedio días order-to-ship: {df['order_to_ship_days'].mean():.1f}")
    shipping_dist = df['shipping_efficiency'].value_counts()
    print(f"   Distribución shipping: {shipping_dist.to_dict()}")
    new_features_created.extend(['order_to_ship_days', 'shipping_efficiency'])
except Exception as e:
    print(f"⚠️ Error creando Order Timing features: {e}")

# Feature 5: Customer Segmentation Score
try:
    # Combinar múltiples factores para scoring
    df['customer_value_score'] = (
        (df['clv_simple'] / df['clv_simple'].max()) * 0.4 +  # 40% CLV
        (df['order_count'] / df['order_count'].max()) * 0.3 +  # 30% Frecuencia
        (df['total_sales'] / df['total_sales'].max()) * 0.3    # 30% Volumen
    )
    
    # Segmentación automática
    df['customer_segment_vdrer'] = pd.cut(df['customer_value_score'], 
                                         bins=[0, 0.3, 0.6, 1.0], 
                                         labels=['Bronze', 'Silver', 'Gold'])
    
    print("✅ customer_segmentation: Segmentación automática basada en valor")
    segment_dist = df['customer_segment_vdrer'].value_counts()
    print(f"   Distribución: {segment_dist.to_dict()}")
    new_features_created.extend(['customer_value_score', 'customer_segment_vdrer'])
except Exception as e:
    print(f"⚠️ Error creando Customer Segmentation: {e}")

print(f"\n✅ FASE 4 COMPLETADA - {len(new_features_created)} features creadas")
print(f"📊 Features: {new_features_created}")

# 1. TEST: Customer Segment tiene diferencias significativas en CLV
if("customer_segment_vdrer" %in% names(superstore) && "clv_simple" %in% names(superstore)) {
  
  cat("\n📊 TEST 1: ANOVA - CLV por Customer Segment\n")
  
  # Limpiar datos para ANOVA
  anova_data <- superstore %>%
    filter(!is.na(customer_segment_vdrer), !is.na(clv_simple)) %>%
    filter(is.finite(clv_simple))
  
  if(nrow(anova_data) > 0) {
    # ANOVA test
    anova_result <- aov(clv_simple ~ customer_segment_vdrer, data = anova_data)
    anova_summary <- summary(anova_result)
    
    cat("F-statistic:", round(anova_summary[[1]][1,4], 3), "\n")
    cat("p-value:", format(anova_summary[[1]][1,5], scientific = TRUE), "\n")
    
    if(anova_summary[[1]][1,5] < 0.05) {
      cat("✅ RESULTADO: Diferencias significativas entre segmentos (p < 0.05)\n")
      
      # Post-hoc test (Tukey HSD)
      tukey_result <- TukeyHSD(anova_result)
      cat("📈 TUKEY HSD - Diferencias entre grupos:\n")
      significant_pairs <- tukey_result$customer_segment_vdrer[,"p adj"] < 0.05
      if(any(significant_pairs)) {
        sig_comparisons <- rownames(tukey_result$customer_segment_vdrer)[significant_pairs]
        cat("Pares significativos:", paste(sig_comparisons, collapse = ", "), "\n")
      }
    } else {
      cat("❌ RESULTADO: No hay diferencias significativas entre segmentos\n")
    }
  }

    cat("95% CI: [", round(cor_test$conf.int[1], 3), ",", 
        round(cor_test$conf.int[2], 3), "]\n")
    
    if(cor_test$p.value < 0.05) {
      cat("✅ RESULTADO: Correlación significativa (p < 0.05)\n")
      
      # Interpretación del tamaño del efecto
      r_squared <- cor_test$estimate^2

  # Cramér's V para tamaño del efecto
  n <- sum(contingency_table)
  cramers_v <- sqrt(chi_test$statistic / (n * (min(dim(contingency_table)) - 1)))
  cat("Cramér's V:", round(cramers_v, 3), "\n")
  
  if(chi_test$p.value < 0.05) {
    cat("✅ RESULTADO: Asociación significativa entre Region y Segment\n")
    cat("Tamaño del efecto:", ifelse(cramers_v < 0.1, "Pequeño",

  # Preparar datos para clustering (por customer)
  customer_features <- superstore %>%
    group_by(Customer_ID) %>%
    summarise(
      clv = first(clv_simple),
      total_sales = first(total_sales),
      order_count = first(order_count),
      avg_order_value = total_sales / order_count
    ) %>%
    filter(!is.na(clv), is.finite(clv), is.finite(avg_order_value)) %>%
    select(-Customer_ID)
  
  # Escalar variables para clustering
  customer_scaled <- scale(customer_features)
  
  # Determinar número óptimo de clusters (Elbow method)
  wss <- map_dbl(1:10, function(k) {
    kmeans(customer_scaled, centers = k, nstart = 25)$tot.withinss
  })
  
  # Encontrar "elbow"
  optimal_k <- which.min(diff(diff(wss))) + 1
  cat("Número óptimo de clusters (Elbow method):", optimal_k, "\n")
  
  # K-means clustering
  set.seed(42)
  kmeans_result <- kmeans(customer_scaled, centers = optimal_k, nstart = 25)
  
  # Agregar clusters al dataset original
  customer_features$cluster <- kmeans_result$cluster
  
  # Estadísticas por cluster
  cat("\n📊 CUSTOMER CLUSTERS - ESTADÍSTICAS:\n")
  cluster_stats <- customer_features %>%
    group_by(cluster) %>%
    summarise(
      customers = n(),
      avg_clv = round(mean(clv), 2),
      avg_total_sales = round(mean(total_sales), 2),
      avg_orders = round(mean(order_count), 1)
    )
  
  print(cluster_stats)
  
  # Silhouette analysis para validar clustering
  sil_analysis <- silhouette(kmeans_result$cluster, dist(customer_scaled))
  avg_sil_width <- mean(sil_analysis[, 3])
  cat("\nPromedio Silhouette Width:", round(avg_sil_width, 3), "\n")
  
  if(avg_sil_width > 0.5) {
