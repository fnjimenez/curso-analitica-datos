# 🔍 CUSTOMER SALES VDRER - CÓDIGO COMPLETO
# ===================================================================

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from datetime import datetime
import warnings
warnings.filterwarnings('ignore')

print("\n" + "="*70)
print("💼 ACTIVIDAD 6: CUSTOMER SALES DATA - EXTRACCIÓN DE VALOR VDRER")
print("="*70)

# ===== FASE 1: VALIDAR =====
df = pd.read_csv('customer_sales_data.csv')
print(f"✅ Dataset cargado: {df.shape[0]:,} filas × {df.shape[1]} columnas")

# Convertir fechas
df['ORDERDATE'] = pd.to_datetime(df['ORDERDATE'])

# Overview del negocio B2B
print(f"\n🏢 OVERVIEW B2B SALES:")
print(f"📅 Período: {df['ORDERDATE'].min().strftime('%Y-%m-%d')} a {df['ORDERDATE'].max().strftime('%Y-%m-%d')}")
print(f"📋 Órdenes únicas: {df['ORDERNUMBER'].nunique():,}")
print(f"👥 Clientes únicos: {df['CUSTOMERNAME'].nunique():,}")
print(f"📦 Productos únicos: {df['PRODUCTCODE'].nunique():,}")
print(f"💰 Revenue total: ${df['SALES'].sum():,.2f}")

# ===== FASE 2: DESCUBRIR =====
print(f"\n🔍 FASE 2: DESCUBRIR - Variables 'inútiles'")

insights_found = []

# ORDERNUMBER patterns
print(f"\n📊 ANÁLISIS ORDERNUMBER:")
print(f"Rango: {df['ORDERNUMBER'].min()} - {df['ORDERNUMBER'].max()}")

# ===== FASE 3: REUTILIZAR =====
print(f"\n♻️ FASE 3: REUTILIZAR - Extrayendo valor")

# Customer acquisition cohorts
first_order_by_customer = df.groupby('CUSTOMERNAME')['ORDERNUMBER'].min()
df['customer_acquisition_cohort'] = df['CUSTOMERNAME'].map(first_order_by_customer)
df['acquisition_quartile'] = pd.qcut(df['customer_acquisition_cohort'], 
                                   q=4, labels=['Early', 'Growth', 'Scale', 'Recent'])

# Product hierarchy
df['product_prefix'] = df['PRODUCTCODE'].str[:2]

# Deal complexity
order_complexity = df.groupby('ORDERNUMBER')['ORDERLINENUMBER'].max()
df = df.merge(order_complexity.rename('deal_complexity_score'), 
              left_on='ORDERNUMBER', right_index=True)

insights_found.extend([
    "ORDERNUMBER revela customer acquisition timing",
    "PRODUCTCODE estructura codifica categorías",
    "ORDERLINENUMBER indica deal complexity"
])

# ===== FASE 4: ENGINEER =====
print(f"\n🔧 FASE 4: ENGINEER - Creando features B2B")

new_features_created = []

# Customer Lifetime Value B2B
customer_metrics = df.groupby('CUSTOMERNAME').agg({
    'SALES': ['sum', 'count', 'mean'],
    'ORDERDATE': ['min', 'max']
}).round(2)

customer_metrics.columns = ['total_sales', 'order_count', 'avg_order_value', 
                           'first_order_date', 'last_order_date']

customer_metrics['customer_lifetime_days'] = (
    customer_metrics['last_order_date'] - customer_metrics['first_order_date']
).dt.days + 1

customer_metrics['clv_b2b'] = (
    customer_metrics['total_sales'] / customer_metrics['customer_lifetime_days'] * 365
).fillna(customer_metrics['total_sales'])

# Merge back
df = df.merge(customer_metrics[['total_sales', 'order_count', 'clv_b2b']], 
              left_on='CUSTOMERNAME', right_index=True, suffixes=('', '_customer'))

# Territory Performance Index
territory_metrics = df.groupby('TERRITORY').agg({
    'SALES': ['sum', 'count', 'mean'],
    'CUSTOMERNAME': 'nunique'
}).round(2)

territory_metrics.columns = ['territory_total_sales', 'territory_order_count', 
                            'territory_avg_sale', 'territory_customers']

territory_metrics['territory_performance_index'] = (
    (territory_metrics['territory_total_sales'] / territory_metrics['territory_total_sales'].max()) * 0.6 +
    (territory_metrics['territory_customers'] / territory_metrics['territory_customers'].max()) * 0.4
)

df = df.merge(territory_metrics[['territory_performance_index']], 
              left_on='TERRITORY', right_index=True)

# Customer Tiers
df['customer_value_score'] = (
    (df['clv_b2b'] / df['clv_b2b'].max()) * 0.5 +
    (df['order_count'] / df['order_count'].max()) * 0.3 +
    (df['total_sales'] / df['total_sales'].max()) * 0.2
)

df['customer_tier'] = pd.cut(df['customer_value_score'], 
                           bins=[0, 0.25, 0.5, 0.75, 1.0], 
                           labels=['Bronze', 'Silver', 'Gold', 'Platinum'])

new_features_created.extend([
    'clv_b2b', 'territory_performance_index', 'customer_tier', 'deal_complexity_score'
])

print(f"✅ Features creadas: {len(new_features_created)}")

# ===== FASE 5: REPORTAR =====
print(f"\n📋 FASE 5: REPORTAR - Exportando dataset enriquecido")

print(f"\n📊 REPORTE FINAL:")
print(f"   💎 Insights extraídos: {len(insights_found)}")
print(f"   ⚙️ Features creadas: {len(new_features_created)}")
print(f"   📈 Dimensiones finales: {df.shape}")

# Customer tier analysis
if 'customer_tier' in df.columns:
    tier_dist = df['customer_tier'].value_counts()
    print(f"\n👥 Customer Tiers: {tier_dist.to_dict()}")

# Export
try:
    filename = 'customer_sales_value_extracted_python.csv'
    
    # Convert datetime to string for compatibility
    export_df = df.copy()
    for col in export_df.columns:
        if export_df[col].dtype == 'datetime64[ns]':
            export_df[col] = export_df[col].dt.strftime('%Y-%m-%d')
    
    export_df.to_csv(filename, index=False, encoding='utf-8')
    print(f"✅ Guardado como: {filename}")
    
    # Download if in Colab
    try:
        from google.colab import files
        files.download(filename)
        print(f"📥 ¡Archivo descargado automáticamente!")
    except:
        print(f"📝 Archivo listo para descargar manualmente")

except Exception as e:
    print(f"⚠️ Error en exportación: {e}")

print("\n" + "="*70)
print("🎉 CUSTOMER SALES VDRER COMPLETADO")
print("🚀 ¡Dataset listo para Power BI!")
print("="*70)

# ===== R STATISTICAL VALIDATION CUSTOMER SALES =====
library(tidyverse)
library(corrplot)
library(cluster)
library(randomForest)

# Cargar dataset enriquecido
customer_sales <- read_csv("customer_sales_value_extracted_python.csv")

cat("💼 CUSTOMER SALES B2B STATISTICAL VALIDATION\n")
cat("=============================================\n")

# Verificar features VDRER
vdrer_features <- c("clv_b2b", "territory_performance_index", 
                    "customer_tier", "deal_complexity_score")

for(feature in vdrer_features) {
  if(feature %in% names(customer_sales)) {
    cat("✅", feature, "- Type:", class(customer_sales[[feature]])[1], "\n")
  } else {
    cat("❌", feature, "- NOT FOUND\n")
  }
}

# TEST 1: ANOVA - Customer Tier vs CLV
if("customer_tier" %in% names(customer_sales) && "clv_b2b" %in% names(customer_sales)) {
  
  anova_data <- customer_sales %>%
    filter(!is.na(customer_tier), !is.na(clv_b2b), is.finite(clv_b2b))
  
  if(nrow(anova_data) > 0) {
    anova_result <- aov(clv_b2b ~ customer_tier, data = anova_data)
    anova_summary <- summary(anova_result)
    
    cat("\n📊 ANOVA - CLV por Customer Tier:\n")
    cat("F-statistic:", round(anova_summary[[1]][1,4], 3), "\n")
    cat("p-value:", format(anova_summary[[1]][1,5], scientific = TRUE), "\n")
    
    if(anova_summary[[1]][1,5] < 0.05) {
      cat("✅ Diferencias significativas entre customer tiers\n")
    }
  }
}

# TEST 2: Territory Performance Correlation
if("territory_performance_index" %in% names(customer_sales)) {
  
  territory_stats <- customer_sales %>%
    group_by(TERRITORY) %>%
    summarise(
      total_revenue = sum(SALES, na.rm = TRUE),
      performance_index = first(territory_performance_index)
    ) %>%
    filter(!is.na(performance_index))
  
  if(nrow(territory_stats) > 3) {
    cor_test <- cor.test(territory_stats$total_revenue, territory_stats$performance_index)
    
    cat("\n📊 CORRELACIÓN Territory Performance vs Revenue:\n")
    cat("Correlación:", round(cor_test$estimate, 3), "\n")
    cat("p-value:", format(cor_test$p.value, scientific = TRUE), "\n")
    
    if(cor_test$p.value < 0.05) {
      cat("✅ Correlación significativa encontrada\n")
    }
  }
}

# Customer Clustering
if("clv_b2b" %in% names(customer_sales)) {
  
  customer_features <- customer_sales %>%
    group_by(CUSTOMERNAME) %>%
    summarise(
      clv = first(clv_b2b),
      total_sales = sum(SALES),
      order_count = n_distinct(ORDERNUMBER)
    ) %>%
    filter(!is.na(clv), is.finite(clv)) %>%
    select(-CUSTOMERNAME)
  
  # K-means clustering
  if(nrow(customer_features) > 10) {
    customer_scaled <- scale(customer_features)
    
    set.seed(42)
    kmeans_result <- kmeans(customer_scaled, centers = 4, nstart = 25)
    
    cat("\n🎯 CUSTOMER CLUSTERING:\n")
    cat("Clusters creados:", max(kmeans_result$cluster), "\n")
    
    # Silhouette analysis
    sil_analysis <- silhouette(kmeans_result$cluster, dist(customer_scaled))
    avg_sil_width <- mean(sil_analysis[, 3])
    cat("Silhouette Width:", round(avg_sil_width, 3), "\n")
    
    if(avg_sil_width > 0.5) {
      cat("✅ Clustering robusto encontrado\n")
    }
  }
}
