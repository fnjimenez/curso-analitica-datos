# 📂 CARGA ROBUSTA DEL DATASET DE RECURSOS HUMANOS
# ============================================
from google.colab import files
import pandas as pd
from io import StringIO
import csv

print("📂 Selecciona tu archivo CSV desde tu computadora:")
print("💡 Tip: Descarga desde: https://fnjimenez.github.io/curso-analitica-datos/datashets/WA_Fn-UseC_-HR-Employee-Attrition.csv")
uploaded = files.upload()

filename = list(uploaded.keys())[0]
print(f"📄 Archivo detectado: {filename}")

# Procesamiento automático del archivo
raw_bytes = uploaded[filename]
preview = raw_bytes[:3000].decode('utf-8', errors='ignore')

# === DETECCIÓN AUTOMÁTICA DEL DELIMITADOR ===
candidates = [',', ';', '\t']
counts = {}
header_line = preview.splitlines()[0]

for delim in candidates:
    try:
        reader = csv.reader([header_line], delimiter=delim)
        n_fields = len(next(reader))
        counts[delim] = n_fields
    except Exception:
        continue

best_delim = max(counts, key=counts.get)
expected_fields = counts[best_delim]

# === REPARACIÓN AUTOMÁTICA DE LÍNEAS PROBLEMÁTICAS ===
lines = raw_bytes.decode('utf-8', errors='ignore').splitlines()
header = lines[0]
data_lines = lines[1:]

print(f"📋 Campos esperados: {expected_fields}")
print(f"📄 Total de líneas de datos: {len(data_lines)}")

problematic_lines = 0
repaired_lines = []

for i, line in enumerate(data_lines):
    if line.strip():
        try:
            fields = next(csv.reader([line], delimiter=best_delim))
            num_fields = len(fields)

            if num_fields == expected_fields:
                repaired_lines.append(line)
            elif num_fields > expected_fields:
                truncated_fields = fields[:expected_fields]
                repaired_line = best_delim.join(f'"{field}"' if best_delim in field or '"' in field else field
                                              for field in truncated_fields)
                repaired_lines.append(repaired_line)
                problematic_lines += 1
            else:
                while len(fields) < expected_fields:
                    fields.append('')
                repaired_line = best_delim.join(f'"{field}"' if best_delim in field or '"' in field else field
                                              for field in fields)
                repaired_lines.append(repaired_line)
                problematic_lines += 1

        except Exception as e:
            print(f"⚠️ Error en línea {i+1}: {str(e)[:50]}...")
            problematic_lines += 1
            repaired_lines.append(line)

print(f"🔧 Total de líneas problemáticas reparadas: {problematic_lines}")

# Reconstruir archivo reparado en memoria
data_clean = '\n'.join([header] + repaired_lines)

# === CARGAR EL DATASET ===
try:
    df = pd.read_csv(StringIO(data_clean), delimiter=best_delim)
    print(f"✅ Dataset cargado exitosamente con delimitador: '{best_delim}'")
    print(f"📊 Forma del dataset: {df.shape}")
except Exception as e:
    print(f"❌ Error leyendo dataset: {e}")
    df = None

# === LIMPIEZA DE NOMBRES DE COLUMNAS ===
if df is not None:
    df.columns = (
        df.columns.str.strip()
        .str.lower()
        .str.replace(' ', '_', regex=False)
        .str.replace('-', '_', regex=False)
        .str.replace(r'[^a-z0-9_]', '', regex=True)
    )
    print(f"\n🧽 Columnas estandarizadas: {df.shape[1]} columnas")
    print(f"📋 Primeras 5 columnas: {list(df.columns[:5])}")

    print(f"\nResumen de carga:")
    print(f"  • Filas cargadas: {len(df):,}")
    print(f"  • Columnas: {len(df.columns)}")
    print(f"  • Líneas reparadas: {problematic_lines}")

print("\n🎉 ¡Importación completada exitosamente!")

    unique_ratio = unique_count / total_count

    if unique_count == 1:
        print(f"  🔴 {col}: CONSTANTE")
        print(f"      └─ Valor único: {df[col].iloc[0]}")
        constant_vars.append(col)
    elif unique_count <= 3 and col not in ['attrition']:  # Excluir variable objetivo
        print(f"  🟡 {col}: CASI CONSTANTE - {unique_count} valores únicos")
        value_counts = df[col].value_counts()
        print(f"      └─ Distribución: {dict(value_counts.head(3))}")

print("\n📈 2. VARIABLES NUMÉRICAS CON BAJA VARIANZA:")
numeric_cols = df.select_dtypes(include=[np.number]).columns

for col in numeric_cols:
    if col not in constant_vars:  # Ya identificadas
        variance = df[col].var()
        cv = df[col].std() / df[col].mean() if df[col].mean() != 0 else 0
        
        # Coeficiente de variación muy bajo indica poca variabilidad
        if cv < 0.1 and col not in almost_constant_vars:
            print(f"  🟠 {col}: BAJA VARIANZA")
            print(f"      └─ Coef. Variación: {cv:.4f}, Varianza: {variance:.2f}")
            low_variance_vars.append(col)

new_features_created = []
feature_descriptions = {}

print("\n🔧 CREACIÓN DE FEATURES DERIVADAS:")

# Feature 1: Career Progression Rate
print("\n1️⃣ CAREER PROGRESSION RATE (Velocidad de progresión profesional)")
try:
    df['career_progression_rate'] = df['joblevel'] / (df['totalworkingyears'] + 1)
    
    stats = {
        'mean': df['career_progression_rate'].mean(),
        'std': df['career_progression_rate'].std(),
        'min': df['career_progression_rate'].min(),
        'max': df['career_progression_rate'].max()
    }
    
    print(f"  ✅ Creada exitosamente")
    print(f"  📊 Media: {stats['mean']:.3f} | Desv.Est: {stats['std']:.3f}")
    print(f"  📈 Rango: {stats['min']:.3f} - {stats['max']:.3f}")
    print(f"  💡 Interpretación: Valores altos = ascenso profesional rápido")
    
    new_features_created.append('career_progression_rate')
    feature_descriptions['career_progression_rate'] = "Velocidad de progresión: joblevel / (años_experiencia + 1)"
    
except Exception as e:
    print(f"  ❌ Error: {e}")

# Feature 2: Company Loyalty Index
print("\n2️⃣ COMPANY LOYALTY INDEX (Índice de lealtad empresarial)")
try:
    df['company_loyalty_index'] = df['yearsatcompany'] / df['numcompaniesworked'].clip(lower=1)
    
    stats = {
        'mean': df['company_loyalty_index'].mean(),
        'std': df['company_loyalty_index'].std(),
        'min': df['company_loyalty_index'].min(),
        'max': df['company_loyalty_index'].max()
    }
    
    print(f"  ✅ Creada exitosamente")
    print(f"  📊 Media: {stats['mean']:.3f} | Desv.Est: {stats['std']:.3f}")
    print(f"  📈 Rango: {stats['min']:.3f} - {stats['max']:.3f}")
    print(f"  💡 Interpretación: Valores altos = mayor estabilidad/lealtad")
    
    new_features_created.append('company_loyalty_index')
    feature_descriptions['company_loyalty_index'] = "Lealtad: años_en_empresa / num_empresas_trabajadas"
    
except Exception as e:
    print(f"  ❌ Error: {e}")

# Feature 3: Promotion Opportunity Gap
print("\n3️⃣ PROMOTION OPPORTUNITY GAP (Brecha de oportunidad de promoción)")

    df['promotion_opportunity_gap'] = df['yearsincurrentrole'] - df['yearssincelastpromotion']
    
    stats = {
        'mean': df['promotion_opportunity_gap'].mean(),
        'std': df['promotion_opportunity_gap'].std(),
        'min': df['promotion_opportunity_gap'].min(),
        'max': df['promotion_opportunity_gap'].max()
    }
    
    print(f"  ✅ Creada exitosamente")
    print(f"  📊 Media: {stats['mean']:.3f} | Desv.Est: {stats['std']:.3f}")
    print(f"  📈 Rango: {stats['min']:.3f} - {stats['max']:.3f}")
    print(f"  💡 Interpretación: Valores positivos = necesidad de promoción")
    
    new_features_created.append('promotion_opportunity_gap')
    feature_descriptions['promotion_opportunity_gap'] = "Oportunidad: años_rol_actual - años_desde_promoción"
    
except Exception as e:
    print(f"  ❌ Error: {e}")

# Feature 4: Work-Life Balance Risk Score
print("\n4️⃣ WORK-LIFE BALANCE RISK SCORE (Riesgo de balance vida-trabajo)")
try:
    # Combinar factores de riesgo
    overtime_risk = (df['overtime'] == 'Yes').astype(int) * 0.4
    distance_risk = (df['distancefromhome'] / df['distancefromhome'].max()) * 0.3
    worklife_risk = ((4 - df['worklifebalance']) / 3) * 0.3  # Invertir escala
    
    df['worklife_risk_score'] = overtime_risk + distance_risk + worklife_risk
    
    stats = {
        'mean': df['worklife_risk_score'].mean(),
        'std': df['worklife_risk_score'].std(),
        'min': df['worklife_risk_score'].min(),
        'max': df['worklife_risk_score'].max()
    }
    
    print(f"  ✅ Creada exitosamente")
    print(f"  📊 Media: {stats['mean']:.3f} | Desv.Est: {stats['std']:.3f}")
    print(f"  📈 Rango: {stats['min']:.3f} - {stats['max']:.3f}")
    print(f"  💡 Interpretación: Valores altos = mayor riesgo de burnout")
    
    new_features_created.append('worklife_risk_score')
    feature_descriptions['worklife_risk_score'] = "Riesgo: combinación de overtime, distancia y balance"
    
except Exception as e:
    print(f"  ❌ Error: {e}")

print(f"\n✅ FASE 4 COMPLETADA - {len(new_features_created)} nuevas features creadas")
