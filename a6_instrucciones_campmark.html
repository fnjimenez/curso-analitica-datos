constant_vars = []
almost_constant_vars = []
marketing_insights = []

for col in df.columns:
    unique_count = df[col].nunique()
    total_count = len(df)
    
    if unique_count == 1:
        print(f"🔴 {col}: CONSTANTE - 1 valor único = {df[col].iloc[0]}")
        constant_vars.append(col)
        
        # Interpretar valor de negocio
        if col == 'Z_CostContact':

        elif col == 'Z_Revenue':
            marketing_insights.append(f"Z_Revenue={df[col].iloc[0]} indica revenue esperado por conversión")
            
    elif unique_count <= 3 and col not in ['Response']:  # Excluir variable objetivo
        print(f"🟡 {col}: CASI CONSTANTE - {unique_count} valores únicos")
        values_dist = df[col].value_counts().to_dict()
        print(f"   Distribución: {values_dist}")
        almost_constant_vars.append(col)
        
        # Análisis específico para variables de campaña
        if col.startswith('AcceptedCmp'):
            total_accepted = sum([v for k, v in values_dist.items() if k == 1])
            print(f"   📈 Total que aceptaron {col}: {total_accepted} ({total_accepted/total_count*100:.1f}%)")

# Análisis especial para ID (posibles patterns de segmentación oculta)

print("\n⚙️ FASE 3: REUTILIZAR - Extrayendo valor comercial de variables 'inútiles'")

marketing_insights_found = []

# 💎 INSIGHT 1: Z_CostContact como baseline de costos de marketing
print("\n💎 INSIGHT 1: Z_CostContact - Baseline de Costos de Marketing")
try:
    cost_per_contact = df['Z_CostContact'].iloc[0]
    total_customers = len(df)
    total_responders = df['Response'].sum()
    
    # Calcular métricas de costo
    total_campaign_cost = cost_per_contact * total_customers
    cost_per_conversion = total_campaign_cost / total_responders if total_responders > 0 else 0
    
    print(f"💰 Costo por contacto estándar: ${cost_per_contact}")
    print(f"📊 Costo total campaña: ${total_campaign_cost:,.0f}")
    print(f"🎯 Costo por conversión: ${cost_per_conversion:.2f}")
    print(f"💡 Insight: Z_CostContact=${cost_per_contact} es el baseline para calcular ROI de campañas")
    
    marketing_insights_found.append("Z_CostContact revela estructura de costos de marketing estándar")
except Exception as e:
    print(f"⚠️ Error en análisis Z_CostContact: {e}")

# 💎 INSIGHT 2: Z_Revenue como target de ingresos esperados
print("\n💎 INSIGHT 2: Z_Revenue - Target de Ingresos por Conversión")
try:
    revenue_per_conversion = df['Z_Revenue'].iloc[0]
    total_responders = df['Response'].sum()
    projected_revenue = revenue_per_conversion * total_responders
    
    print(f"💵 Revenue esperado por conversión: ${revenue_per_conversion}")
    print(f"📈 Revenue proyectado total: ${projected_revenue:,.0f}")
    print(f"💡 Insight: Z_Revenue=${revenue_per_conversion} es el target de ingresos para medir éxito")
    
    # Calcular ROI potencial
    roi_ratio = revenue_per_conversion / cost_per_contact if 'cost_per_contact' in locals() else 0
    print(f"📊 ROI potencial por contacto: {roi_ratio:.2f}x (Revenue/Cost)")
    
    marketing_insights_found.append("Z_Revenue revela target de ingresos por conversión")
except Exception as e:
    print(f"⚠️ Error en análisis Z_Revenue: {e}")

# 💎 INSIGHT 3: AcceptedCmp2=0 revela falla de campaña específica
print("\n💎 INSIGHT 3: AcceptedCmp2 - Análisis de Falla de Campaña")
try:
    cmp2_acceptance = df['AcceptedCmp2'].sum()
    total_customers = len(df)
    
    print(f"❌ Campaña 2 - Aceptaciones: {cmp2_acceptance} de {total_customers} ({cmp2_acceptance/total_customers*100:.1f}%)")
    
    if cmp2_acceptance == 0:
        print(f"🔍 INSIGHT CRÍTICO: Campaña 2 tuvo 0% de aceptación")
        print(f"💡 Posibles causas:")
        print(f"   • Targeting incorrecto")
        print(f"   • Oferta no atractiva")
        print(f"   • Canal de comunicación inadecuado")
        print(f"   • Timing inadecuado")
        
        # Comparar con otras campañas
        campaign_performance = {}
        for col in ['AcceptedCmp1', 'AcceptedCmp3', 'AcceptedCmp4', 'AcceptedCmp5']:
            if col in df.columns:
                acceptance_rate = df[col].sum() / total_customers * 100
                campaign_performance[col] = acceptance_rate
                print(f"   📊 {col}: {acceptance_rate:.1f}% aceptación")
    
    marketing_insights_found.append("AcceptedCmp2=0 revela falla completa de campaña específica")
except Exception as e:
    print(f"⚠️ Error en análisis AcceptedCmp2: {e}")

# 💎 INSIGHT 4: Patterns en Customer ID (cohortes de adquisición)
print("\n💎 INSIGHT 4: Customer ID - Cohortes de Adquisición Ocultas")
try:
    # Analizar patrones de respuesta por rangos de ID
    id_ranges = pd.qcut(df['ID'], q=4, labels=['Early_Customers', 'Growth_Phase', 'Expansion_Era', 'Recent_Customers'])
    df['customer_cohort'] = id_ranges
    
    cohort_analysis = df.groupby('customer_cohort').agg({
        'Response': ['count', 'sum', 'mean'],
        'Income': 'mean'
    }).round(3)
    
    print(f"🏢 Análisis por cohorte de clientes:")
    for cohort in df['customer_cohort'].unique():
        cohort_data = df[df['customer_cohort'] == cohort]
        response_rate = cohort_data['Response'].mean()
        avg_income = cohort_data['Income'].mean()
        customer_count = len(cohort_data)
        
        print(f"   • {cohort}: {customer_count} clientes, {response_rate:.1%} respuesta, Income promedio: ${avg_income:,.0f}")
    
    marketing_insights_found.append("Customer ID revela cohortes con diferentes comportamientos de respuesta")
except Exception as e:
    print(f"⚠️ Error en análisis Customer ID: {e}")

# ⚙️ FASE 4: ENGINEER - Crear Features Derivadas de Marketing

print("\n🔧 FASE 4: ENGINEER - Creando features de marketing con valor comercial")

new_marketing_features = []

# Feature 1: Customer Lifetime Value Proxy
try:
    # Calcular spending total por cliente
    spending_cols = ['MntWines', 'MntFruits', 'MntMeatProducts', 'MntFishProducts', 'MntSweetProducts', 'MntGoldProds']
    df['total_spending'] = df[spending_cols].sum(axis=1)
    
    # Calcular age actual (asumiendo año actual 2024)
    current_year = 2024
    df['age'] = current_year - df['Year_Birth']
    
    # CLV Proxy basado en spending e income
    df['clv_proxy'] = (df['total_spending'] * df['age']) / (df['age'] - 18 + 1)  # Normalizar por años como adulto
    
    print("✅ clv_proxy: Customer Lifetime Value aproximado")
    print(f"   Media: ${df['clv_proxy'].mean():.2f}")
    print(f"   Interpretación: Más alto = mayor valor potencial del cliente")
    new_marketing_features.append('clv_proxy')
except Exception as e:
    print(f"⚠️ Error creando clv_proxy: {e}")

# Feature 2: Purchase Frequency Index
try:
    # Calcular frecuencia de compra por canal
    purchase_cols = ['NumWebPurchases', 'NumCatalogPurchases', 'NumStorePurchases']
    df['total_purchases'] = df[purchase_cols].sum(axis=1)
    
    # Normalizar por deals (ofertas aprovechadas)
    df['purchase_frequency_index'] = df['total_purchases'] / (df['NumDealsPurchases'] + 1)  # +1 para evitar división por 0
    
    print("\n✅ purchase_frequency_index: Índice de frecuencia de compra")
    print(f"   Media: {df['purchase_frequency_index'].mean():.3f}")
    print(f"   Interpretación: Más alto = compra más frecuentemente sin necesidad de ofertas")
    new_marketing_features.append('purchase_frequency_index')
except Exception as e:
    print(f"⚠️ Error creando purchase_frequency_index: {e}")

# Feature 3: Income-Spending Ratio (Capacidad de gasto vs gasto real)
try:
    df['income_spending_ratio'] = df['Income'] / (df['total_spending'] + 1)  # +1 para evitar división por 0
    
    print("\n✅ income_spending_ratio: Ratio ingresos vs gasto")
    print(f"   Media: {df['income_spending_ratio'].mean():.3f}")
    print(f"   Interpretación: Más alto = mayor capacidad de gasto no aprovechada")
    new_marketing_features.append('income_spending_ratio')
except Exception as e:
    print(f"⚠️ Error creando income_spending_ratio: {e}")

# Feature 4: Channel Preference Score
try:
    # Calcular preferencia de canal basada en proporción de compras
    df['web_preference'] = df['NumWebPurchases'] / (df['total_purchases'] + 1)
    df['catalog_preference'] = df['NumCatalogPurchases'] / (df['total_purchases'] + 1)
    df['store_preference'] = df['NumStorePurchases'] / (df['total_purchases'] + 1)
    
    # Score combinado de preferencia digital vs física
    df['digital_preference_score'] = (df['web_preference'] + df['catalog_preference']) - df['store_preference']
    
    print("\n✅ digital_preference_score: Preferencia por canales digitales")
    print(f"   Media: {df['digital_preference_score'].mean():.3f}")
    print(f"   Interpretación: Positivo = prefiere digital, Negativo = prefiere tienda física")
    new_marketing_features.append('digital_preference_score')
except Exception as e:
    print(f"⚠️ Error creando digital_preference_score: {e}")

# Feature 5: Family Structure Index
try:
    # Combinar kids y teens en un índice de estructura familiar
    df['family_size'] = df['Kidhome'] + df['Teenhome']
    
    # Crear categorías familiares
    df['family_structure'] = pd.cut(df['family_size'], 
                                   bins=[-1, 0, 1, 3], 
                                   labels=['No_Kids', 'Small_Family', 'Large_Family'])
    
    # Score numérico para análisis
    family_mapping = {'No_Kids': 0, 'Small_Family': 1, 'Large_Family': 2}
    df['family_structure_score'] = df['family_structure'].map(family_mapping)
    
    print("\n✅ family_structure_score: Índice de estructura familiar")
    print(f"   Distribución: {df['family_structure'].value_counts().to_dict()}")
    print(f"   Interpretación: 0=Sin hijos, 1=Familia pequeña, 2=Familia grande")
    new_marketing_features.append('family_structure_score')
except Exception as e:
    print(f"⚠️ Error creando family_structure_score: {e}")

# Feature 6: Response Likelihood Score (basado en histórico de campañas)
try:
    # Sumar todas las campañas aceptadas previamente
    campaign_cols = ['AcceptedCmp1', 'AcceptedCmp3', 'AcceptedCmp4', 'AcceptedCmp5']
    df['historical_campaign_acceptance'] = df[campaign_cols].sum(axis=1)
    
    # Normalizar por número de campañas
    df['campaign_responsiveness'] = df['historical_campaign_acceptance'] / len(campaign_cols)
    
    print("\n✅ campaign_responsiveness: Índice de respuesta a campañas")
    print(f"   Media: {df['campaign_responsiveness'].mean():.3f}")
    print(f"   Interpretación: Más alto = más propenso a responder a campañas")
    new_marketing_features.append('campaign_responsiveness')
except Exception as e:
    print(f"⚠️ Error creando campaign_responsiveness: {e}")

print(f"\n✅ FASE 4 COMPLETADA - {len(new_marketing_features)} nuevas features de marketing creadas")
print(f"📊 Features: {new_marketing_features}")

# Análisis de correlación con Response de nuevas features
print(f"\n📈 CORRELACIÓN CON RESPONSE (nuevas features de marketing):")
if 'new_marketing_features' in locals():
    # Crear variable numérica de response
    df['response_numeric'] = df['Response'].astype(int)
    
    correlations = {}
    for feature in new_marketing_features:
        if feature in df.columns and df[feature].dtype in ['float64', 'int64']:
            try:
                corr = df[feature].corr(df['response_numeric'])
                correlations[feature] = corr
                print(f"   • {feature}: {corr:.3f}")
            except:
                print(f"   • {feature}: No se pudo calcular correlación")
    
    # Identificar feature más predictiva
    if correlations:
        best_feature = max(correlations.items(), key=lambda x: abs(x[1]))

# Preparar datos para clustering (solo variables numéricas relevantes)
cluster_vars <- marketing_enriched %>%
  select(Income, total_spending, age, clv_proxy, 
         purchase_frequency_index, digital_preference_score) %>%
  na.omit() %>%
  scale()

# Determinar número óptimo de clusters
set.seed(42)
wss <- map_dbl(1:10, ~{
  k_means_result <- kmeans(cluster_vars, centers = .x, nstart = 20)
  k_means_result$tot.withinss
})

# Visualizar elbow method
elbow_plot <- tibble(k = 1:10, wss = wss) %>%
  ggplot(aes(k, wss)) +
  geom_line() +
  geom_point() +
  labs(title = "📈 Elbow Method: Optimal Number of Customer Clusters",
       x = "Number of Clusters (k)",
       y = "Within-Cluster Sum of Squares") +
  theme_minimal()

print(elbow_plot)

# Ejecutar K-means con 4 clusters (basado en elbow method)
optimal_clusters <- 4
k_means_result <- kmeans(cluster_vars, centers = optimal_clusters, nstart = 20)

# Agregar clusters al dataset original
marketing_enriched$customer_cluster <- as.factor(k_means_result$cluster)

# Test chi-cuadrado para factores de éxito de campaigns
education_response <- table(marketing_enriched$Education, marketing_enriched$Response)
education_test <- chisq.test(education_response)

print(education_test)

# Cramér's V para effect size
n <- sum(education_response)
cramers_v_education <- sqrt(education_test$statistic / (n * (min(dim(education_response)) - 1)))

# Preparar datos para modelo logístico
model_data <- marketing_enriched %>%
  select(Response, Income, age, total_spending, clv_proxy, 
         purchase_frequency_index, digital_preference_score,
         Education, Marital_Status) %>%
  na.omit() %>%

    Education = as.factor(Education),
    Marital_Status = as.factor(Marital_Status)
  )

# Dividir en train/test
set.seed(42)
train_index <- createDataPartition(model_data$Response, p = 0.8, list = FALSE)
train_data <- model_data[train_index, ]
test_data <- model_data[-train_index, ]

# Entrenar modelo logístico
logistic_model <- glm(Response ~ ., 
                     data = train_data, 
                     family = binomial)

cat("\n=== 📊 LOGISTIC REGRESSION SUMMARY ===\n")
print(summary(logistic_model))

# Evaluar modelo
predictions <- predict(logistic_model, test_data, type = "response")
predicted_classes <- ifelse(predictions > 0.5, 1, 0)
actual_classes <- as.numeric(as.character(test_data$Response))

# Confusion matrix
confusion_matrix <- table(Predicted = predicted_classes, Actual = actual_classes)
accuracy <- sum(diag(confusion_matrix)) / sum(confusion_matrix)

cat("\n=== 🎯 MODEL PERFORMANCE ===\n")
print(confusion_matrix)
cat("Accuracy:", round(accuracy, 3), "\n")
